# SchemaParser: turns the tree of an already parsed @stxt.schema document into a
# Schema object. The input document MUST have "Schema (@stxt.schema): <ns>" as root.
#
# Precondition: the document has already been parsed by the core parser and,
# optionally, validated against the meta-schema (@stxt.schema). Here the logical
# representation is built.

# The schema language is written with inline nodes: "Schema:", "Node:", "Child:", "Type:"...
# Anything else is not a schema. Ports call this before reading value or children of a node
# of the schema document, so a misuse fails with a typed error, never with a class cast.
FUNCTION inline(node: Node): InlineNode
    IF node IS InlineNode
        RETURN node
    END IF
    THROW ValidationException(node.getLine(), "INVALID_SCHEMA", "Node '" + node.getName() + "' must be inline in a schema")
END FUNCTION

# Turns the root node of a schema document into a Schema.
FUNCTION transformNodeToSchema(node: Node): Schema

    node_name: STRING        = node.getCanonicalName()
    namespace_schema: STRING = node.getNamespace()

    # The root MUST be "schema" in the @stxt.schema namespace
    IF node_name != "schema" OR namespace_schema != Schema.SCHEMA_NAMESPACE
        THROW ValidationException(node.getLine(), "NOT_STXT_SCHEMA",
            "Expected schema(" + Schema.SCHEMA_NAMESPACE + ") but got " + node_name + "(" + namespace_schema + ")")
    END IF
    root: InlineNode = inline(node)

    # Optional description of the schema
    description: STRING = NULL
    description_node: Node = root.getChild("description")
    IF description_node IS NOT NULL
        description = description_node.getText()
    END IF

    # The inline value of the root Schema node is the target namespace
    schema: Schema = Schema(root.getValue(), root.getLine(), description)

    # Canonical names of the declared Nodes (to check §8.1 below)
    all_names: STRING[] = []

    # Each "Node" child of the schema defines a NodeDefinition
    FOREACH n IN root.getChildren("node")
        sch_node: NodeDefinition = createNodeDefinition(n, schema.getNamespace())
        schema.addNodeDefinition(sch_node)
        all_names.push(sch_node.getCanonicalName())
    END FOREACH

    # Every Child of the schema's own namespace MUST refer to a Node declared in this
    # schema (cross-namespace Children are validated in their own schema, not here).
    FOREACH sch_node IN schema.getNodes().values()
        FOREACH sch_child IN sch_node.getChildren().values()
            IF sch_child.getNamespace() == schema.getNamespace()
                IF NOT all_names.contains(sch_child.getCanonicalName())
                    THROW ValidationException(0, "CHILD_NOT_DEFINED",
                        "Child " + sch_child.getCanonicalName() + " not defined in " + schema.getNamespace())
                END IF
            END IF
        END FOREACH
    END FOREACH

    RETURN schema
END FUNCTION

# Builds a NodeDefinition from a "Node:" entry.
FUNCTION createNodeDefinition(node: Node, namespace: STRING): NodeDefinition

    n: InlineNode = inline(node)

    # The inline value of "Node" is the name of the target node
    name: STRING = n.getValue()

    # Default type INLINE when no "Type:" is given
    type: STRING = "INLINE"
    type_node: Node = n.getChild("type")
    IF type_node IS NOT NULL
        type = type_node.getText()
    END IF

    # Optional description of the node
    description: STRING = NULL
    description_node: Node = n.getChild("description")
    IF description_node IS NOT NULL
        description = description_node.getText()
    END IF

    result: NodeDefinition = NodeDefinition(name, type, n.getLine(), description)

    # Children declared inside "Children:"
    children: Node = n.getChild("children")
    IF children IS NOT NULL
        # Schema error §13.5: Children in a Node whose type does not admit children
        IF NOT TypeRegistry.admitsChildren(type)
            THROW ValidationException(children.getLine(), "CHILDREN_NOT_ALLOWED_FOR_TYPE",
                "Type " + type + " does not allow children (node " + name + ")")
        END IF

        FOREACH child IN inline(children).getChildren("child")
            putChildToNodeDefinition(result, child, namespace)
        END FOREACH
    END IF

    # Allowed values: only valid for the ENUM type
    values: Node[] = n.getChildren("values")
    IF values.size() > 0

        IF type != "ENUM"
            THROW ValidationException(n.getLine(), "VALUES_ONLY_SUPPORTED_BY_ENUM",
                "Values only supported for type ENUM, not for type " + type)
        END IF

        IF values.size() > 1
            THROW RuntimeException("INVALID_SIZE_VALUES", "Unexpected number of values: " + values.size())
        END IF

        # Collect every "Value:" inside the "Values:" node
        values_node: Node = values[0]
        values = inline(values_node).getChildren("value")
        FOREACH value IN values
            result.addValue(value.getText(), value.getLine())
        END FOREACH
    END IF

    # An ENUM MUST declare at least one value
    IF type == "ENUM" AND values.size() == 0
        THROW ValidationException(n.getLine(), "VALUES_EMPTY_FOR_ENUM", "ENUM Type must include values")
    END IF

    RETURN result
END FUNCTION

# Builds a ChildDefinition from a "Child:" entry and adds it to the NodeDefinition.
PROCEDURE putChildToNodeDefinition(node_definition: NodeDefinition, child_node: Node, def_namespace: STRING)

    child: InlineNode = inline(child_node)

    # The inline value of "Child" is the name of the child, with an optional namespace in ()
    nn: NameNamespace = NameNamespaceParser.parse(child.getValue(), def_namespace, child.getLine(), child.getValue())
    name: STRING      = nn.getName()
    namespace: STRING = nn.getNamespace()

    min: INTEGER = getInteger(child, "min")
    max: INTEGER = getInteger(child, "max")

    # Invalid cardinality when Min > Max (STXT-SCHEMA-SPEC §10 and §13.7)
    IF min IS NOT NULL AND max IS NOT NULL AND min > max
        THROW ValidationException(child.getLine(), "MIN_GREATER_THAN_MAX", "Min " + min + " greater than Max " + max)
    END IF

    sch_child: ChildDefinition = ChildDefinition(name, namespace, min, max, child.getLine())

    node_definition.addChildDefinition(sch_child)
END PROCEDURE

# Reads an integer child (Min/Max) of a node. Returns NULL when it is not there.
FUNCTION getInteger(node: InlineNode, child_name: STRING): INTEGER
    n: Node = node.getChild(child_name)
    IF n IS NULL
        RETURN NULL
    END IF

    # Fail loudly when the value is not a valid integer
    IF NOT isInteger(n.getText())
        THROW ValidationException(node.getLine(), "INVALID_INTEGER", "Integer not valid: " + n.getText())
    END IF

    RETURN parseInteger(n.getText())
END FUNCTION
