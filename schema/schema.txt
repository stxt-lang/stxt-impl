# Schema: logical representation of a @stxt.schema associated with a target namespace.
# It holds a set of node definitions (NodeDefinition), indexed by their canonical name.
# It is the result of SchemaParser.transformNodeToSchema() (or of compiling a template).

CLASS Schema

    # Namespace of the schema language itself
    CONSTANT SCHEMA_NAMESPACE: STRING = "@stxt.schema"

    FIELD namespace: STRING
    FIELD description: STRING       # optional description of the schema, or NULL

    # Key = canonical name of the node. Preserves insertion order.
    FIELD nodes: MAP<STRING, NodeDefinition> = {}

    CONSTRUCTOR (namespace, line, description)
        namespace   = lowerCase(namespace)
        description = description

        # The target namespace MUST be a valid namespace ('a.b' minimum, ASCII)
        validateNamespaceFormat(namespace, line)
    END CONSTRUCTOR

    FUNCTION getNamespace(): STRING
        RETURN namespace
    END FUNCTION

    FUNCTION getDescription(): STRING
        RETURN description
    END FUNCTION

    FUNCTION getNodes(): MAP<STRING, NodeDefinition>
        RETURN nodes
    END FUNCTION

    # Looks up a node definition by name (canonicalized before the lookup).
    # NULL if the node is not defined in this schema.
    FUNCTION getNodeDefinition(name: STRING): NodeDefinition
        RETURN nodes[normalizeChars(name)]
    END FUNCTION

    PROCEDURE addNodeDefinition(node_definition: NodeDefinition)
        qname: STRING = node_definition.getCanonicalName()

        # Two Node definitions cannot share a canonical name (schema error)
        IF nodes.containsKey(qname)
            THROW ValidationException(0, "NODE_DEF_ALREADY_DEFINED", "Exists a previous node definition with: " + qname)
        END IF

        nodes.put(qname, node_definition)
    END PROCEDURE

END CLASS
