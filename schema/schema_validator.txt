# SchemaValidator: validates an STXT node against the schema of its namespace.
# It implements the Validator interface of the parser core (see processors/validator.txt),
# so it can be plugged into the parse (loosely coupled, streaming validation) or run
# afterwards over the tree.
#
# It resolves the schema through a SchemaProvider and checks: existence of the node in
# the schema, validity of the value according to its type, the closed content model
# (every child must be declared) and the cardinalities of the declared children.
#
# Following the Validator contract, every method RETURNS the errors found instead of
# throwing them, so a caller can collect the errors of a whole document.

CLASS SchemaValidator IMPLEMENTS Validator

    FIELD schema_provider: SchemaProvider
    FIELD recursive_validation: BOOLEAN = false

    CONSTRUCTOR (schema_provider)
        schema_provider = schema_provider
    END CONSTRUCTOR

    CONSTRUCTOR (schema_provider, recursive)
        schema_provider      = schema_provider
        recursive_validation = recursive
    END CONSTRUCTOR

    # Entry point (Validator.validate). Validates the node and, optionally, its subtree.
    FUNCTION validate(node: Node): ValidationException[]

        errors: ValidationException[] = []

        namespace: STRING = node.getNamespace()
        sch: Schema       = schema_provider.getSchema(namespace)

        IF sch IS NULL
            errors.push(ValidationException(node.getLine(), "SCHEMA_NOT_FOUND", "Not found schema: " + namespace))
            RETURN errors
        END IF

        errors.pushAll(validateAgainstSchema(node, sch))

        # Optional recursive validation of the children (only an InlineNode has any)
        IF recursive_validation AND node IS InlineNode
            FOREACH n IN node.getChildren()
                errors.pushAll(validate(n))
            END FOREACH
        END IF

        RETURN errors
    END FUNCTION

    # Validates a node against an already resolved schema: existence, value type,
    # declared children and cardinalities.
    FUNCTION validateAgainstSchema(node: Node, sch: Schema): ValidationException[]

        errors: ValidationException[] = []

        schema_node: NodeDefinition = sch.getNodeDefinition(node.getCanonicalName())

        # Closed model: the node MUST be defined in the schema of its namespace
        IF schema_node IS NULL
            errors.push(ValidationException(node.getLine(), "NODE_NOT_EXIST_IN_SCHEMA",
                "NOT EXIST NODE " + node.getCanonicalName() + " for namespace " + sch.getNamespace()))
            RETURN errors
        END IF

        errors.pushAll(validateValue(schema_node, node))
        errors.pushAll(validateChildrenDeclared(schema_node, node))
        errors.pushAll(validateCount(schema_node, node))

        RETURN errors
    END FUNCTION


    # The children of a node for the purposes of the content model: a TextNode has none
    FUNCTION childrenOf(node: Node): Node[]
        IF node IS InlineNode
            RETURN node.getChildren()
        END IF
        RETURN []
    END FUNCTION

    # Closed content model (STXT-SCHEMA-SPEC §6): only the direct children declared in
    # the definition of the parent are allowed; with no Children declared, none is.
    # The error is reported on the line of the offending child.
    FUNCTION validateChildrenDeclared(ns_node: NodeDefinition, node: Node): ValidationException[]

        errors: ValidationException[] = []

        FOREACH child IN childrenOf(node)
            IF NOT ns_node.getChildren().containsKey(child.getQualifiedName())
                errors.push(ValidationException(child.getLine(), "CHILD_NOT_DECLARED",
                    "Child '" + child.getQualifiedName() + "' not declared in node '" + node.getQualifiedName() + "'"))
            END IF
        END FOREACH

        RETURN errors
    END FUNCTION

    # Validates the shape/content of the value according to the declared type.
    # The types throw (see schema/types.txt); the exception is collected here.
    FUNCTION validateValue(ns_node: NodeDefinition, n: Node): ValidationException[]

        errors: ValidationException[] = []

        node_type: STRING = ns_node.getType()

        type: Type = TypeRegistry.get(node_type)
        IF type IS NULL
            errors.push(ValidationException(n.getLine(), "TYPE_NOT_SUPPORTED", "Node type not supported: " + node_type))
            RETURN errors
        END IF

        TRY
            type.validate(ns_node, n)
        CATCH ve: ValidationException
            errors.push(ve)
        CATCH e
            errors.push(ValidationException(n.getLine(), "VALIDATION_ERROR", e.getMessage()))
        END TRY

        RETURN errors
    END FUNCTION

    # Counts the direct children by qualified name and checks cardinalities.
    # The validation is order independent: it counts occurrences, not positions.
    FUNCTION validateCount(ns_node: NodeDefinition, node: Node): ValidationException[]

        errors: ValidationException[] = []

        count: MAP<STRING, INTEGER> = {}
        children_by_name: MAP<STRING, Node[]> = {}

        FOREACH child IN childrenOf(node)
            child_name: STRING = child.getQualifiedName()

            IF count.containsKey(child_name)
                count[child_name] = count[child_name] + 1
            ELSE
                count[child_name] = 1
                children_by_name[child_name] = []
            END IF

            children_by_name[child_name].push(child)
        END FOREACH

        # For every declared child, check min/max against the observed number
        FOREACH ch_node IN ns_node.getChildren().values()
            qname: STRING = ch_node.getQualifiedName()

            observed: INTEGER = 0
            IF count.containsKey(qname)
                observed = count[qname]
            END IF

            observed_children: Node[] = []
            IF children_by_name.containsKey(qname)
                observed_children = children_by_name[qname]
            END IF

            errors.pushAll(validateCardinality(ch_node, observed, node, observed_children))
        END FOREACH

        RETURN errors
    END FUNCTION

    FUNCTION validateCardinality(ch_node: ChildDefinition, num: INTEGER, node: Node, children: Node[]): ValidationException[]

        errors: ValidationException[] = []

        min: INTEGER = ch_node.getMin()
        max: INTEGER = ch_node.getMax()

        IF min IS NOT NULL AND num < min
            errors.push(ValidationException(node.getLine(), "INVALID_NUMBER",
                num + " nodes of '" + ch_node.getQualifiedName() + "' and min is " + min))
        END IF

        IF max IS NOT NULL AND num > max
            # Error on the parent
            errors.push(ValidationException(node.getLine(), "INVALID_NUMBER",
                num + " nodes of '" + ch_node.getQualifiedName() + "' and max is " + max))

            # And an error on each offending child, for line-accurate reporting
            FOREACH child IN children
                errors.push(ValidationException(child.getLine(), "INVALID_NUMBER",
                    "Too many '" + ch_node.getQualifiedName() + "' nodes: found " + num + ", max is " + max))
            END FOREACH
        END IF

        RETURN errors
    END FUNCTION

END CLASS
