# NodeDefinition: definition of a node inside a schema. Equivalent to one "Node:"
# entry of the schema document. It holds the value type, the expected children
# (ChildDefinition), the optional description and, for ENUM, the allowed values.

CLASS NodeDefinition

    FIELD name: STRING
    FIELD normalized_name: STRING
    FIELD type: STRING
    FIELD description: STRING       # optional description of the node, or NULL

    # Key = qualified name of the child (namespace:canonical_name)
    FIELD children: MAP<STRING, ChildDefinition> = {}

    # Allowed values for the ENUM type. Set semantics: no duplicates.
    FIELD values: STRING[] = []

    CONSTRUCTOR (name, type, line, description)
        name            = compactSpaces(name)
        normalized_name = normalizeChars(name)
        type            = type
        description     = description

        IF isEmpty(normalized_name)
            THROW ValidationException(line, "INVALID_NODE_NAME", "Node name not valid: " + name)
        END IF
    END CONSTRUCTOR

    FUNCTION getName(): STRING
        RETURN name
    END FUNCTION

    FUNCTION getNormalizedName(): STRING
        RETURN normalized_name
    END FUNCTION

    FUNCTION getType(): STRING
        RETURN type
    END FUNCTION

    FUNCTION getDescription(): STRING
        RETURN description
    END FUNCTION

    # Sets the optional description (used by the template "Description >>" block).
    PROCEDURE setDescription(description: STRING)
        description = description
    END PROCEDURE

    FUNCTION getChildren(): MAP<STRING, ChildDefinition>
        RETURN children
    END FUNCTION

    PROCEDURE addChildDefinition(child_definition: ChildDefinition)
        qname: STRING = child_definition.getQualifiedName()

        # Two equivalent Child entries (same canonical name + namespace) are an error
        IF children.containsKey(qname)
            THROW ValidationException(0, "CHILD_DEF_ALREADY_DEFINED", "Exists a previous node definition with: " + qname)
        END IF

        children.put(qname, child_definition)
    END PROCEDURE

    # Adds an allowed value (ENUM). Duplicated values (after trim) are an error
    # (STXT-SCHEMA-SPEC §13.9 / STXT-TEMPLATE-SPEC §14.14). Same code as
    # ChildLineParser: it is the same condition coming through the other entry point.
    PROCEDURE addValue(value: STRING, line: INTEGER)
        value = trimToNotNull(value)

        IF values.contains(value)
            THROW ValidationException(line, "VALUE_DUPLICATED", "The values " + value + " is duplicated")
        END IF

        values.push(value)
    END PROCEDURE

    # true if no restricted values are defined, or if the value is among the allowed ones
    FUNCTION isAllowedValue(value: STRING): BOOLEAN
        IF values.size() == 0
            RETURN true
        END IF
        RETURN values.contains(value)
    END FUNCTION

    FUNCTION getValues(): STRING[]
        RETURN values
    END FUNCTION

END CLASS
