# SchemaProvider: resolves the Schema that applies to a namespace.
#
# CONTRACT: getSchema() returns the Schema, or NULL when the provider has no schema
# for that namespace. Providers do NOT throw "not found" errors: it is the consumer
# (typically SchemaValidator) who reports SCHEMA_NOT_FOUND when the resolution ends
# in NULL. This keeps providers composable (fallback chains, discovery).
#
# Composable implementations: SchemaProviderMemory (registered schemas + parent
# fallback), SchemaProviderMeta (the embedded meta-schema), DiscoveryResult (see
# discovery/), TemplateSchemaProviderMemory (see template/).

# -------------------------------------------------------------------
# Interface
# -------------------------------------------------------------------
INTERFACE SchemaProvider
    # The schema of the namespace, or NULL if this provider has none for it.
    FUNCTION getSchema(namespace: STRING): Schema
END INTERFACE

# -------------------------------------------------------------------
# In-memory provider with parent fallback
# -------------------------------------------------------------------
# Keeps the schemas added with addSchema() indexed by namespace, and falls back to a
# parent provider (the meta-schema provider by default) for the namespaces it does
# not know.
CLASS SchemaProviderMemory IMPLEMENTS SchemaProvider

    FIELD parent: SchemaProvider
    FIELD schemas: MAP<STRING, Schema> = {}     # key = lowercased namespace, insertion order

    CONSTRUCTOR ()
        parent = SchemaProviderMeta()
    END CONSTRUCTOR

    CONSTRUCTOR (parent)
        parent = parent
    END CONSTRUCTOR

    # Resolves from the registered schemas, delegating to the parent when unknown here.
    FUNCTION getSchema(namespace: STRING): Schema
        key: STRING = lowerCase(namespace)

        result: Schema = schemas[key]
        IF result IS NULL
            result = parent.getSchema(namespace)
        END IF

        RETURN result
    END FUNCTION

    # Parses a schema document, validates it against the meta-schema and registers it
    # under its own namespace.
    PROCEDURE addSchema(text: STRING)
        parser: Parser = Parser()
        nodes: Node[] = parser.parse(text)

        IF nodes.length() != 1
            THROW ValidationException(0, "INVALID_SCHEMA", "There are " + nodes.length() + ", and expected is 1")
        END IF

        node: Node = nodes[0]

        # The document MUST validate against the meta-schema (@stxt.schema)
        validator: SchemaValidator = SchemaValidator(SchemaProviderMeta(), true)
        errors: ValidationException[] = validator.validate(node)
        IF errors.length() > 0
            THROW errors[0]
        END IF

        schema: Schema = SchemaParser.transformNodeToSchema(node)
        schemas.put(schema.getNamespace(), schema)
    END PROCEDURE

    # Removes every schema registered here (the parent is left untouched).
    PROCEDURE clear()
        schemas = {}
    END PROCEDURE

    # Every schema registered here, in registration order.
    FUNCTION getAllSchemas(): Schema[]
        RETURN schemas.values()
    END FUNCTION

END CLASS

# -------------------------------------------------------------------
# Provider of the embedded meta-schema (@stxt.schema)
# -------------------------------------------------------------------
# Defines IN CODE the meta-schema of the schema language itself, so that a loaded
# schema can validate itself (bootstrap). Its content is normative:
# STXT-SCHEMA-SPEC §15.2 ("the schema of the schema").
CLASS SchemaProviderMeta IMPLEMENTS SchemaProvider

    # META_TEXT is the following @stxt.schema document, embedded verbatim in code:
    #
    #   Schema (@stxt.schema): @stxt.schema
    #       Node: Schema
    #           Children:
    #               Child: Description
    #                   Max: 1
    #               Child: Node
    #                   Min: 1
    #       Node: Node
    #           Children:
    #               Child: Type
    #                   Max: 1
    #               Child: Children
    #                   Max: 1
    #               Child: Description
    #                   Max: 1
    #               Child: Values
    #                   Max: 1
    #       Node: Children
    #           Type: GROUP
    #           Children:
    #               Child: Child
    #                   Min: 1
    #       Node: Description
    #           Type: TEXT
    #       Node: Child
    #           Children:
    #               Child: Min
    #                   Max: 1
    #               Child: Max
    #                   Max: 1
    #       Node: Min
    #           Type: NATURAL
    #       Node: Max
    #           Type: NATURAL
    #       Node: Type
    #           Type: ENUM
    #           Values:
    #               Value: INLINE
    #               Value: BLOCK
    #               Value: TEXT
    #               Value: BOOLEAN
    #               Value: URL
    #               Value: INTEGER
    #               Value: NATURAL
    #               Value: NUMBER
    #               Value: DATE
    #               Value: TIME
    #               Value: TIMESTAMP
    #               Value: UUID
    #               Value: EMAIL
    #               Value: HEXADECIMAL
    #               Value: BINARY
    #               Value: BASE64
    #               Value: GROUP
    #               Value: ENUM
    #               Value: MARKDOWN
    #       Node: Values
    #           Type: GROUP
    #           Children:
    #               Child: Value
    #                   Min: 1
    #       Node: Value
    CONSTANT META_TEXT: STRING

    FIELD meta: Schema

    CONSTRUCTOR ()
        parser: Parser = Parser()
        nodes: Node[] = parser.parse(META_TEXT)

        IF nodes.length() != 1
            THROW ValidationException(0, "META_SCHEMA_INVALID", "Meta schema must produce exactly 1 document, got " + nodes.length())
        END IF

        meta = SchemaParser.transformNodeToSchema(nodes[0])
    END CONSTRUCTOR

    # Serves the meta-schema; NULL for any other namespace (SchemaProvider contract).
    FUNCTION getSchema(namespace: STRING): Schema
        IF namespace != Schema.SCHEMA_NAMESPACE
            RETURN NULL
        END IF

        RETURN meta
    END FUNCTION

END CLASS
