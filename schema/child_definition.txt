# ChildDefinition: declaration of an expected child inside a Node ("Child:").
# It defines the logical pair (canonical name + effective namespace) and the min/max
# cardinality. min and max are optional: NULL means "no limit set".

CLASS ChildDefinition

    FIELD name: STRING
    FIELD canonical_name: STRING
    FIELD namespace: STRING
    FIELD min: INTEGER          # NULL = no minimum (effectively 0)
    FIELD max: INTEGER          # NULL = no maximum (unbounded)

    CONSTRUCTOR (name, namespace, min, max, line)
        name            = compactSpaces(name)
        canonical_name = normalizeChars(name)
        namespace       = lowerCase(namespace)
        min             = min
        max             = max

        # The namespace of the child (inherited or explicit) MUST be valid
        validateNamespaceFormat(namespace, line)

        # A Child name is a node name too, so apply the same full STXT name rules
        # as the core parser and NodeDefinition.
        IF NOT isValidNodeName(name)
            THROW ValidationException(line, "INVALID_NODE_NAME", "Node name not valid: " + name)
        END IF
    END CONSTRUCTOR

    FUNCTION getName(): STRING
        RETURN name
    END FUNCTION

    FUNCTION getCanonicalName(): STRING
        RETURN canonical_name
    END FUNCTION

    FUNCTION getNamespace(): STRING
        RETURN namespace
    END FUNCTION

    FUNCTION getMin(): INTEGER
        RETURN min
    END FUNCTION

    FUNCTION getMax(): INTEGER
        RETURN max
    END FUNCTION

    # Logical identity of the child: namespace:canonical_name, or just the name if
    # there is no namespace. Used as the key in NodeDefinition.getChildren().
    FUNCTION getQualifiedName(): STRING
        IF isEmpty(namespace)
            RETURN canonical_name
        ELSE
            RETURN namespace + ":" + canonical_name
        END IF
    END FUNCTION

END CLASS
