# DiscoveryFileSystem: minimal file-system abstraction used by DiscoveryResolver
# (STXT-DISCOVERY-SPEC).
#
# The resolver treats paths as OPAQUE STRINGS: it never parses or concatenates them
# itself, so an implementation may back them with plain OS paths, editor URIs or an
# in-memory tree for tests. All paths returned by an implementation MUST be canonical
# enough that string equality means "same directory".
#
# Platform note: hosts with asynchronous I/O (e.g. JavaScript) implement these
# operations as async; the resolver logic is the same either way.
#
# Symlink note (STXT-DISCOVERY-SPEC §3, §10): an implementation over a real file system
# SHOULD NOT follow directory symbolic links, so the resolver's recursive descent cannot
# be lured into a symlink loop or into an unrelated tree. The recommended way is to OMIT
# directory symlinks from the listing returned by listDirectory (equivalently, resolve
# is_directory without following links so a symlinked directory is not recursed into).
# The descent is additionally bounded by DEFAULT_MAX_DESCENT in the resolver, independent
# of the adapter, and listDirectory failures are tolerated per-directory (collectFilesAt),
# so neither a loop nor an unreadable directory can turn resolution into unbounded
# recursion or an escaping exception.

# An entry of a directory listing, as returned by listDirectory().
CLASS DiscoveryEntry
    FIELD path: STRING          # full path, in the same form the file system uses everywhere
    FIELD name: STRING          # base name (last path segment)
    FIELD is_directory: BOOLEAN
END CLASS

INTERFACE DiscoveryFileSystem

    # true if the path exists and is a directory; false otherwise (including I/O errors).
    FUNCTION isDirectory(path: STRING): BOOLEAN

    # The immediate entries of a directory, in any order.
    FUNCTION listDirectory(path: STRING): DiscoveryEntry[]

    # Reads a file as UTF-8 text.
    FUNCTION readFile(path: STRING): STRING

    # The parent directory of a path, or NULL when the path is the file-system root.
    FUNCTION parentOf(path: STRING): STRING

    # Joins a directory path and a child name.
    FUNCTION join(path: STRING, name: STRING): STRING

END INTERFACE
