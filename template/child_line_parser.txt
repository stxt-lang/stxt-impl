# ChildLineParser: parses the optional RuleSpec of a "Structure >>" line.
# The RuleSpec is the part AFTER the ':' of a template line, i.e. what the core parser
# returns as the inline value of the node. Its logical shape is:
#
#     [ "(" cardinality ")" ]  [ type | "@Reference" ]  [ "[" ENUM values "]" ]
#
# This function does NOT know the name or the namespace of the node (the core parser
# resolves those from the node itself); it only interprets cardinality, type and values.

# The RuleSpec is split into its three optional parts by a hand-written scan, NOT by a
# regular expression. Until 2026-09-06 every port used the pattern
#     ^[ \t]*(?:\([ \t]*(count)[ \t]*\)[ \t]*)?(type)?(?:\[[ \t]*(values)[ \t]*\][ \t]*)?[ \t]*$
# whose blank runs around the lazy groups backtracked in O(n³) on a line without the
# closing ']' (a Structure line of 10 000 characters, within the parser's line limit, took
# minutes in every port; a template reaches the parser through discovery, so the line is
# not necessarily the user's own). Blanks here are U+0020/U+0009 only, like everywhere else
# in the language (STXT-SPEC §4 and TEMPLATE-SPEC §6.2/§9): the platform's trim and \s
# would also match non-ASCII whitespace and are deliberately avoided, so every port accepts
# and rejects exactly the same lines.
#
# Shape of the RuleSpec, with blanks allowed between the parts:
#
#     [ "(" count ")" ]  [ type ]  [ "[" values "]" ]
#
# Returns NULL when the line does not have that shape (STRUCTURE_LINE_NOT_VALID); otherwise
# the three parts, each NULL when absent and already trimmed when present. The rules the
# old pattern enforced are kept exactly, so that no line changes meaning:
#   - count runs from an opening "(" (only leading blanks before it) to the FIRST ")"
#     after it. Once trimmed it MUST NOT be empty and MUST NOT start with "(" — "()",
#     "( )" and "((1))" are not valid lines. Anything else is allowed inside: a bad count
#     is CARDINALITY_NOT_VALID, not STRUCTURE_LINE_NOT_VALID.
#   - type is everything after the count up to the first "[" (or the end), and MUST NOT
#     contain "(", ")" or "]" — "TEXT (1)", "(1) (2)" and "TEXT a]" are not valid lines.
#   - values runs from that "[" to the FIRST "]" after it (a "[" inside is content; no
#     "]" is not a valid line), and after the "]" only blanks may follow — "[a] b" and
#     "[a] [b]" are not valid lines.
FUNCTION splitRuleSpec(raw_line: STRING): (count: STRING, type: STRING, values: STRING) | NULL
    n: INTEGER = raw_line.length()
    i: INTEGER = 0
    WHILE i < n AND isBlank(raw_line[i])
        i = i + 1
    END WHILE

    count: STRING = NULL
    IF i < n AND raw_line[i] == "("
        close: INTEGER = raw_line.indexOf(")", i + 1)
        IF close == -1
            RETURN NULL
        END IF
        count = StringUtils.trim(raw_line.substring(i + 1, close))
        IF count.isEmpty() OR count[0] == "("
            RETURN NULL
        END IF
        i = close + 1
    END IF

    open: INTEGER = raw_line.indexOf("[", i)
    type_end: INTEGER = (open == -1) ? n : open
    type: STRING = raw_line.substring(i, type_end)
    IF type.contains("(") OR type.contains(")") OR type.contains("]")
        RETURN NULL
    END IF
    type = StringUtils.trim(type)
    IF type.isEmpty()
        type = NULL
    END IF

    values: STRING = NULL
    IF open != -1
        close: INTEGER = raw_line.indexOf("]", open + 1)
        IF close == -1
            RETURN NULL
        END IF
        values = StringUtils.trim(raw_line.substring(open + 1, close))
        FOR j = close + 1 TO n - 1
            IF NOT isBlank(raw_line[j])
                RETURN NULL
            END IF
        END FOR
    END IF

    RETURN (count, type, values)
END FUNCTION

FUNCTION parse(raw_line: STRING, line_number: INTEGER): ChildLine

    # Line with no RuleSpec: everything defaults
    IF StringUtils.trim(raw_line).isEmpty()
        RETURN ChildLine(NULL, NULL, NULL, NULL)
    END IF

    parts = splitRuleSpec(raw_line)
    IF parts IS NULL
        THROW ValidationException(line_number, "STRUCTURE_LINE_NOT_VALID", "Line not valid: " + raw_line)
    END IF

    # --- Type (or @Name reference) ---
    type: STRING = parts.type                       # NULL when absent, trimmed otherwise

    # --- Cardinality (STXT-TEMPLATE-SPEC §7.1) ---
    count: STRING = parts.count
    IF count IS NULL
        count = ""
    END IF

    min: INTEGER = NULL
    max: INTEGER = NULL

    IF count.isEmpty() OR count == "*"
        # (*) or absent: no limits
        min = NULL
        max = NULL

    ELSE IF count == "?"
        # zero or one
        min = NULL
        max = 1

    ELSE IF count == "+"
        # one or more
        min = 1
        max = NULL

    ELSE IF endsWith(count, "+")
        # num+ : num or more
        min = parseCount(count.substring(0, count.length() - 1), count, raw_line, line_number)
        max = NULL

    ELSE IF count.indexOf(",") != -1
        # min,max
        parts: STRING[] = split(count, ",")
        IF parts.length() != 2
            THROW ValidationException(line_number, "CARDINALITY_NOT_VALID", "Invalid count " + count + " in line: " + raw_line)
        END IF

        min = parseCount(StringUtils.trim(parts[0]), count, raw_line, line_number)
        max = parseCount(StringUtils.trim(parts[1]), count, raw_line, line_number)

        # Invalid cardinality when min > max (§7.1)
        IF min > max
            THROW ValidationException(line_number, "MIN_GREATER_THAN_MAX",
                "Min " + min + " greater than Max " + max + " in line: " + raw_line)
        END IF

    ELSE IF endsWith(count, "-")
        # num- : up to num
        min = NULL
        max = parseCount(count.substring(0, count.length() - 1), count, raw_line, line_number)

    ELSE
        # exact num : min = max = num
        min = parseCount(count, count, raw_line, line_number)
        max = min
    END IF

    # --- ENUM values ---
    # Brackets being present (even empty ones, "[]") count as an explicit definition of
    # values: a non-NULL (possibly empty) array is returned, to tell it apart from
    # having no brackets at all (values stays NULL).
    values: STRING[] = NULL
    values_str: STRING = parts.values
    IF values_str IS NOT NULL
        values = []
        # split MUST keep the trailing empty item ("[a, b,]" gives three parts), as the
        # JavaScript and Python split do; in Java that is split(",", -1).
        parts: STRING[] = split(values_str, ",")
        FOREACH part IN parts
            part = StringUtils.trim(part)
            # An empty item ("[a, , b]", "[a, b,]") is an error, as an empty Value: is in a
            # schema (STXT-TEMPLATE-SPEC §14.14). Only the whole list may be empty ("[]"),
            # which the template parser reports as VALUES_REQUIRED.
            IF part.isEmpty() AND parts.size() > 1
                THROW ValidationException(line_number, "VALUE_EMPTY", "Empty ENUM value in " + values_str)
            END IF
            IF part.isEmpty()
                CONTINUE
            END IF

            # ENUM values cannot repeat after the trim (§14.14)
            IF values.contains(part)
                THROW ValidationException(line_number, "VALUE_DUPLICATED", "The values " + part + " is duplicated")
            END IF
            values.push(part)
        END FOREACH
    END IF

    RETURN ChildLine(type, min, max, values)
END FUNCTION

# num, min and max must be NON-NEGATIVE integers, with no trailing text, and must not
# exceed MAX_CARDINALITY = 4294967295 = 2^32 - 1 (§7.1) — the same bound as Min/Max in a
# schema. See getInteger in schema/schema_parser.txt for the overflow note.
FUNCTION parseCount(num: STRING, count: STRING, raw_line: STRING, line_number: INTEGER): INTEGER
    IF NOT isNatural(num)
        THROW ValidationException(line_number, "CARDINALITY_NOT_VALID", "Invalid count " + count + " in line: " + raw_line)
    END IF
    # A numeral with more digits than MAX_CARDINALITY has (10) exceeds the bound whatever
    # its value: rejected before converting, so a platform that limits the digits of a
    # string-to-integer conversion (Python raises past 4 300 digits) never gets to raise.
    IF num.length() > MAX_CARDINALITY_DIGITS
        THROW ValidationException(line_number, "CARDINALITY_NOT_VALID", "Invalid count " + count + " in line: " + raw_line)
    END IF
    value: INTEGER = parseInteger(num)
    IF value > MAX_CARDINALITY
        THROW ValidationException(line_number, "CARDINALITY_NOT_VALID", "Invalid count " + count + " in line: " + raw_line)
    END IF
    RETURN value
END FUNCTION
