# ChildLineParser: parses the optional RuleSpec of a "Structure >>" line.
# The RuleSpec is the part AFTER the ':' of a template line, i.e. what the core parser
# returns as the inline value of the node. Its logical shape is:
#
#     [ "(" cardinality ")" ]  [ type | "@Reference" ]  [ "[" ENUM values "]" ]
#
# This function does NOT know the name or the namespace of the node (the core parser
# resolves those from the node itself); it only interprets cardinality, type and values.

# Pattern with three optional groups: count, type, values.
CONSTANT CHILD_LINE_PATTERN: REGEX =
    "^\\s*" +
    "(?:\\(\\s*(?<count>[^()\\s][^)]*?)\\s*\\)\\s*)?" +     # (cardinality)
    "(?<type>[^()\\[\\]]*)?" +                              # type or @reference
    "(?:\\[\\s*(?<values>[^]]*?)\\s*\\]\\s*)?" +            # [ENUM values]
    "\\s*$"

FUNCTION parse(raw_line: STRING, line_number: INTEGER): ChildLine

    # Line with no RuleSpec: everything defaults
    IF raw_line.trim().isEmpty()
        RETURN ChildLine(NULL, NULL, NULL, NULL)
    END IF

    matcher = CHILD_LINE_PATTERN.matcher(raw_line)
    IF NOT matcher.matches()
        THROW ValidationException(line_number, "INVALID_CHILD_LINE", "Line not valid: " + raw_line)
    END IF

    # --- Type (or @Name reference) ---
    type: STRING = matcher.group("type")
    IF type IS NOT NULL
        type = type.trim()
    END IF
    IF type IS NULL OR type.isEmpty()
        type = NULL
    END IF

    # --- Cardinality (STXT-TEMPLATE-SPEC §7.1) ---
    count: STRING = matcher.group("count")
    IF count IS NULL
        count = ""
    END IF
    count = count.trim()

    min: INTEGER = NULL
    max: INTEGER = NULL

    IF count.isEmpty() OR count == "*"
        # (*) or absent: no limits
        min = NULL
        max = NULL

    ELSE IF count == "?"
        # zero or one
        min = NULL
        max = 1

    ELSE IF count == "+"
        # one or more
        min = 1
        max = NULL

    ELSE IF endsWith(count, "+")
        # num+ : num or more
        min = parseCount(count.substring(0, count.length() - 1), count, raw_line, line_number)
        max = NULL

    ELSE IF count.indexOf(",") != -1
        # min,max
        parts: STRING[] = split(count, ",")
        IF parts.length() != 2
            THROW ValidationException(line_number, "INVALID_CHILD_COUNT", "Invalid count " + count + " in line: " + raw_line)
        END IF

        min = parseCount(parts[0].trim(), count, raw_line, line_number)
        max = parseCount(parts[1].trim(), count, raw_line, line_number)

        # Invalid cardinality when min > max (§7.1)
        IF min > max
            THROW ValidationException(line_number, "MIN_GREATER_THAN_MAX",
                "Min " + min + " greater than Max " + max + " in line: " + raw_line)
        END IF

    ELSE IF endsWith(count, "-")
        # num- : up to num
        min = NULL
        max = parseCount(count.substring(0, count.length() - 1), count, raw_line, line_number)

    ELSE
        # exact num : min = max = num
        min = parseCount(count, count, raw_line, line_number)
        max = min
    END IF

    # --- ENUM values ---
    # Brackets being present (even empty ones, "[]") count as an explicit definition of
    # values: a non-NULL (possibly empty) array is returned, to tell it apart from
    # having no brackets at all (values stays NULL).
    values: STRING[] = NULL
    values_str: STRING = matcher.group("values")
    IF values_str IS NOT NULL
        values = []
        parts: STRING[] = split(values_str, ",")
        FOREACH part IN parts
            part = part.trim()
            IF part.isEmpty()
                CONTINUE
            END IF

            # ENUM values cannot repeat after the trim (§14.14)
            IF values.contains(part)
                THROW ValidationException(line_number, "VALUE_DUPLICATED", "The values " + part + " is duplicated")
            END IF
            values.push(part)
        END FOREACH
    END IF

    RETURN ChildLine(type, min, max, values)
END FUNCTION

# num, min and max must be NON-NEGATIVE integers, with no trailing text (§7.1).
FUNCTION parseCount(num: STRING, count: STRING, raw_line: STRING, line_number: INTEGER): INTEGER
    IF NOT isNatural(num)
        THROW ValidationException(line_number, "INVALID_CHILD_COUNT", "Invalid count " + count + " in line: " + raw_line)
    END IF
    RETURN parseInteger(num)
END FUNCTION
