# TemplateParser: turns a @stxt.template document into an equivalent Schema.
# A template is "syntactic sugar" over a schema (STXT-TEMPLATE-SPEC §13): the
# "Structure >>" block describes the expected structure, and it is compiled here into
# NodeDefinition / ChildDefinition. Where a template and a schema express the same
# condition, the SAME error code is used through both entry points.
#
# Strategy: the content of the "Structure >>" block is, itself, an STXT document. It
# is re-parsed with the core parser and walked depth first, producing one node
# definition per local line and one Child per child line. Errors thrown while working
# on the re-parsed block are re-thrown with the line shifted by the block offset, so
# they point at the real line of the original document.

# Turns the root node "Template (@stxt.template): <ns>" into a Schema.
FUNCTION transformNodeToSchema(node: Node): Schema

    # The inline value of "Template" is the target namespace
    result: Schema = Schema(node.getValue(), node.getLine(), NULL)

    # Locate the mandatory "Structure >>" block
    structure: Node = node.getChild("structure")
    IF structure IS NULL
        THROW ValidationException(node.getLine(), "TEMPLATE_STRUCTURE_REQUIRED", "Template must define 'Structure >>'")
    END IF

    # The text of the block is an STXT document: re-parse it.
    text: STRING    = structure.getText()
    offset: INTEGER = structure.getLine()

    TRY
        nodes: Node[] = Parser().parse(text)

        # Every top-level node of Structure goes into the schema
        FOREACH n IN nodes
            addToSchema(result, n)
        END FOREACH
    CATCH pe: ParseException
        # Re-throw with the line shifted to the one of the original document.
        # (ValidationException extends ParseException: implementations that tell them
        # apart MUST preserve the subtype so the severity is not downgraded.)
        THROW pe.withLine(pe.getLine() + offset)
    END TRY

    # Optional "Description >>" block: descriptions for the defined nodes (§12)
    description: Node = node.getChild("description")
    IF description IS NOT NULL
        TRY
            nodes: Node[] = Parser().parse(description.getText())
            addDescriptions(result, nodes)
        CATCH pe: ParseException
            THROW pe.withLine(pe.getLine() + description.getLine())
        END TRY
    END IF

    RETURN result
END FUNCTION

# Adds to the schema the definition a Structure node declares, along with its children.
PROCEDURE addToSchema(schema: Schema, node: Node)

    namespace: STRING = node.getNamespace()
    name: STRING      = node.getName()

    # Parse the RuleSpec (cardinality / type / values) from the inline value
    cl: ChildLine = ChildLineParser.parse(node.getValue(), node.getLine())

    # No explicit namespace => the target namespace of the template
    IF isEmpty(namespace)
        namespace = schema.getNamespace()
    END IF

    # Cross-namespace node: NOT defined locally; it may only declare cardinality —
    # no type, no ENUM values and no children (§6.4, §10 and §14.15).
    IF namespace != schema.getNamespace()
        type: STRING = cl.getType()
        IF type IS NOT NULL AND NOT type.trim().isEmpty()
            THROW ValidationException(node.getLine(), "TYPE_DEFINITION_NOT_ALLOWED",
                "Not allowed type definition in external namespaces")
        END IF

        IF cl.getValues() IS NOT NULL
            THROW ValidationException(node.getLine(), "VALUES_NOT_ALLOWED_IN_EXTERNAL_NAMESPACE",
                "Not allowed values in external namespaces (node " + node.getName() + ")")
        END IF

        IF node.getChildren().size() > 0
            THROW ValidationException(node.getLine(), "CHILDREN_NOT_ALLOWED_IN_EXTERNAL_NAMESPACE",
                "Not allowed children in external namespaces (node " + node.getName() + ")")
        END IF

        RETURN     # no definitions are created for nodes of other namespaces
    END IF

    # New definition or a reappearance (reference)?
    schema_node: NodeDefinition = schema.getNodeDefinition(name)

    IF schema_node IS NULL
        # --- New definition ---
        type: STRING = cl.getType()
        IF type IS NULL
            type = "INLINE"
        END IF

        # A "@" here means a reference that resolved to nothing: at this point the
        # schema already holds both the previous (closed) definitions and the open
        # ancestors (§6.4 and §14.11).
        IF type.startsWith("@")
            THROW ValidationException(node.getLine(), "REFERENCE_NOT_FOUND",
                "Reference '" + type + "' does not point to a previous definition or an open ancestor")
        END IF

        schema_node = NodeDefinition(node.getName(), type, node.getLine(), NULL)
        schema.addNodeDefinition(schema_node)

        IF TypeRegistry.get(type) IS NULL
            THROW ValidationException(node.getLine(), "TYPE_NOT_VALID", "Type not valid: " + type)
        END IF

        # ENUM values, if any
        values: STRING[] = cl.getValues()
        IF values IS NOT NULL
            IF type != "ENUM"
                # Same code as SchemaParser: a template is sugar equivalent to a schema
                THROW ValidationException(node.getLine(), "VALUES_ONLY_SUPPORTED_BY_ENUM",
                    "Values only supported for type ENUM, not for type " + type)
            END IF

            FOREACH value IN values
                schema_node.addValue(value, node.getLine())
            END FOREACH
        END IF

        # An ENUM with no list of values is an invalid template (§9 and §13.7)
        IF type == "ENUM" AND (values IS NULL OR values.length() == 0)
            THROW ValidationException(node.getLine(), "VALUES_EMPTY_FOR_ENUM", "ENUM Type must include values")
        END IF

    ELSE
        # --- Reappearance: MUST be a "@Node Name" reference ---
        type: STRING = cl.getType()

        # A reappearance without "@" would redefine an existing node: error.
        IF type IS NULL OR NOT type.startsWith("@")
            THROW ValidationException(node.getLine(), "NODE_DEFINED_MULTIPLE_TIMES",
                "Multiple node reference must start with @: " + node.getName())
        END IF

        reference: STRING = type.substring(1).trim()

        # Reference and explicit type on the same line (§14.13)
        explicit_type: STRING = referenceType(reference, node.getNormalizedName())
        IF explicit_type IS NOT NULL
            THROW ValidationException(node.getLine(), "REFERENCE_WITH_TYPE_NOT_ALLOWED",
                "Reference '@" + node.getName() + "' can not declare a type: " + explicit_type)
        END IF

        # The name of the reference MUST match (canonically) the one of the line (§14.12)
        IF normalizeChars(reference) != node.getNormalizedName()
            THROW ValidationException(node.getLine(), "NODE_REFERENCE_NOT_VALID",
                "Reference must be '@" + node.getName() + "', not '" + reference + "'")
        END IF

        # A reference may override the cardinality, but it may redefine neither the
        # ENUM values nor the children (§6.4)
        IF cl.getValues() IS NOT NULL
            THROW ValidationException(node.getLine(), "VALUES_NOT_ALLOWED_IN_REFERENCE",
                "Reference '@" + node.getName() + "' can not redefine ENUM values")
        END IF

        IF node.getChildren().size() > 0
            THROW ValidationException(node.getLine(), "CHILDREN_NOT_ALLOWED_IN_REFERENCE",
                "Reference '@" + node.getName() + "' can not redefine children")
        END IF

        RETURN     # valid reference: nothing is redefined, no children are processed
    END IF

    # The definition exists: process its direct children.
    children: Node[] = node.getChildren()

    # Template error §14.9: children under an effective type that does not admit them
    IF children.size() > 0 AND NOT TypeRegistry.admitsChildren(schema_node.getType())
        THROW ValidationException(node.getLine(), "CHILDREN_NOT_ALLOWED_FOR_TYPE",
            "Type " + schema_node.getType() + " does not allow children (node " + node.getName() + ")")
    END IF

    FOREACH child IN children

        cl = ChildLineParser.parse(child.getValue(), child.getLine())

        child_name: STRING      = child.getName()
        child_namespace: STRING = child.getNamespace()
        IF isEmpty(child_namespace)
            child_namespace = schema.getNamespace()
        END IF

        # The child is declared as a Child (with its cardinality) in the current definition
        sch_child: ChildDefinition = ChildDefinition(child_name, child_namespace,
            cl.getMin(), cl.getMax(), child.getLine())
        schema_node.addChildDefinition(sch_child)

        # And processed recursively as a definition/reference
        addToSchema(schema, child)
    END FOREACH
END PROCEDURE

# Tells "@Node Name TYPE" (reference + type, error §14.13) apart from "@Another Name"
# (reference with a different name, error §14.12). Since node names may contain
# spaces, the only reliable reading is: if the last token is a known type and what
# comes before it is the name of the node itself, the line is declaring both things.
# Returns the declared type, or NULL when the reference carries no type.
FUNCTION referenceType(reference: STRING, normalized_name: STRING): STRING

    cut: INTEGER = reference.lastIndexOf(" ")
    IF cut < 0
        RETURN NULL
    END IF

    candidate: STRING = reference.substring(cut + 1).trim()
    rest: STRING      = reference.substring(0, cut)

    IF TypeRegistry.get(candidate) IS NOT NULL AND normalizeChars(rest) == normalized_name
        RETURN candidate
    END IF

    RETURN NULL
END FUNCTION

# Attaches to the node definitions the descriptions declared in "Description >>" (§12).
# The content of the block is an STXT document: one "Node Name: description" (or a
# block) per described node.
PROCEDURE addDescriptions(schema: Schema, nodes: Node[])

    FOREACH node IN nodes

        namespace: STRING = node.getNamespace()
        IF isEmpty(namespace)
            namespace = schema.getNamespace()
        END IF

        # No external descriptions
        IF namespace != schema.getNamespace()
            THROW ValidationException(node.getLine(), "EXTERNAL_DESCRIPTION_NOT_ALLOWED",
                "Not allowed description in external namespaces")
        END IF

        # No children either
        IF node.getChildren().size() > 0
            THROW ValidationException(node.getLine(), "CHILDREN_DESCRIPTION_NOT_ALLOWED",
                "Not allowed children in description")
        END IF

        # The described node MUST exist in the schema
        node_def: NodeDefinition = schema.getNodeDefinition(node.getName())
        IF node_def IS NULL
            THROW ValidationException(node.getLine(), "NODE_NOT_FOUND",
                "Not found node with name: " + node.getName())
        END IF

        # At most one description per node (§12)
        IF node_def.getDescription() IS NOT NULL
            THROW ValidationException(node.getLine(), "DESCRIPTION_ALREADY_DEFINED",
                "Exists a previous description for node: " + node.getName())
        END IF

        node_def.setDescription(node.getText())
    END FOREACH
END PROCEDURE
