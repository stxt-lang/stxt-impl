# Template providers: obtain a Schema from @stxt.template documents. They reuse the
# SchemaProvider interface of the schema package, since a template compiles to the same
# logical representation (Schema) as a schema.

# -------------------------------------------------------------------
# In-memory provider fed with template documents
# -------------------------------------------------------------------
# Each template is turned into its equivalent Schema and registered under its own
# namespace. Inherits getSchema()/clear()/getAllSchemas() from SchemaProviderMemory.
CLASS TemplateSchemaProviderMemory EXTENDS SchemaProviderMemory

    CONSTRUCTOR ()
        super(MetaTemplateSchemaProvider())
    END CONSTRUCTOR

    CONSTRUCTOR (parent)
        super(parent)
    END CONSTRUCTOR

    # Parses a template document, validates it against the template meta-schema and
    # registers the schema it produces.
    PROCEDURE addTemplate(template: STRING)
        parser: Parser = Parser()
        nodes: Node[] = parser.parse(template)

        IF nodes.length() != 1
            THROW ValidationException(0, "TEMPLATE_MULTIPLE_ROOTS", "There are " + nodes.length() + ", and expected is 1")
        END IF

        node: Node = nodes[0]

        # The document MUST validate against the meta-template (@stxt.template)
        validator: SchemaValidator = SchemaValidator(MetaTemplateSchemaProvider(), true)
        errors: ValidationException[] = validator.validate(node)
        IF errors.length() > 0
            THROW errors[0]
        END IF

        # Compile the template to its equivalent Schema
        schema: Schema = TemplateParser.transformNodeToSchema(node)

        IF isEmpty(schema.getNamespace())
            THROW ValidationException(node.getLine(), "TEMPLATE_NAMESPACE_EMPTY", "Template namespace is empty")
        END IF

        schemas.put(schema.getNamespace(), schema)
    END PROCEDURE

END CLASS

# -------------------------------------------------------------------
# Provider of the embedded meta-template (@stxt.template)
# -------------------------------------------------------------------
# Defines IN CODE the meta-schema of the template language itself, so that a loaded
# template can validate itself (bootstrap). Its content is normative:
# STXT-TEMPLATE-SPEC §16 ("the template of the template"). It validates the shape of
# any @stxt.template document:
#   Template (@stxt.template):
#       Description: (?) TEXT
#       Structure:   (1) BLOCK
CLASS MetaTemplateSchemaProvider IMPLEMENTS SchemaProvider

    # META_TEXT is the following @stxt.template document, embedded verbatim in code:
    #
    #   Template (@stxt.template): @stxt.template
    #       Structure >>
    #           Template (@stxt.template):
    #               Description: (?) TEXT
    #               Structure: (1) BLOCK
    CONSTANT META_TEXT: STRING

    FIELD meta: Schema

    CONSTRUCTOR ()
        parser: Parser = Parser()
        nodes: Node[] = parser.parse(META_TEXT)

        IF nodes.length() != 1
            THROW ValidationException(0, "META_SCHEMA_INVALID", "Meta schema must produce exactly 1 document, got " + nodes.length())
        END IF

        # The meta-template itself is compiled with the TemplateParser
        meta = TemplateParser.transformNodeToSchema(nodes[0])
    END CONSTRUCTOR

    # Serves the meta-template; NULL for any other namespace (SchemaProvider contract).
    FUNCTION getSchema(namespace: STRING): Schema
        IF namespace != "@stxt.template"
            RETURN NULL
        END IF

        RETURN meta
    END FUNCTION

END CLASS
