# ChildLine: the result of parsing the RuleSpec of a "Structure >>" line.
# An immutable container with the cardinality (min/max), the type and, for ENUM, the
# list of values. Any field may be NULL when it was not specified.
#
# Examples of RuleSpec producing a ChildLine:
#   (1)                -> min=1, max=1, type=NULL,  values=NULL
#   (?) TEXT           -> min=NULL, max=1, type="TEXT"
#   (1) ENUM [a, b]    -> min=1, max=1, type="ENUM", values=["a","b"]
#   @Body Content      -> type="@Body Content" (reference to a node)
#
# Note: values=[] (empty but not NULL) means the brackets were present but empty
# ("[]"), which counts as an explicit definition of values (see child_line_parser.txt).

CLASS ChildLine

    FIELD type: STRING          # type, "@Name" reference or NULL
    FIELD min: INTEGER          # NULL = no minimum
    FIELD max: INTEGER          # NULL = no maximum
    FIELD values: STRING[]      # ENUM values, or NULL when no brackets were present

    CONSTRUCTOR (type, min, max, values)
        type   = type
        min    = min
        max    = max
        values = values
    END CONSTRUCTOR

    FUNCTION getType(): STRING
        RETURN type
    END FUNCTION

    FUNCTION getMin(): INTEGER
        RETURN min
    END FUNCTION

    FUNCTION getMax(): INTEGER
        RETURN max
    END FUNCTION

    FUNCTION getValues(): STRING[]
        RETURN values
    END FUNCTION

END CLASS
