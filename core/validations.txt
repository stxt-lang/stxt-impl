# Format of a logical namespace (STXT-SPEC §7):
#   * Lower-case ASCII letters, digits and dot only.
#   * It may optionally start with '@' (reserved namespaces).
#   * Two or more domain-style labels separated by '.': label := [a-z0-9]+
#   * Valid examples: "a.b", "com.example.docs", "@stxt.schema".
#
# Checked by a hand-written scan, NOT by the regex ^@?[a-z0-9]+(\.[a-z0-9]+)+$ every port
# used until 2026-09-06: a regex engine that implements a repeated group by recursion
# (java.util.regex) overflowed the stack with ~2 000 labels, a 4 000-character line well
# within the line limit, and a stack overflow is not an exception the parser can turn into
# an error. The scan is linear and identical in every port.
FUNCTION isValidNamespaceFormat(namespace: STRING): BOOLEAN
    n: INTEGER = namespace.length()
    i: INTEGER = (n > 0 AND namespace[0] == "@") ? 1 : 0
    labels: INTEGER = 0
    LOOP
        start: INTEGER = i
        WHILE i < n AND isLabelChar(namespace[i])       # [a-z0-9], ASCII only
            i = i + 1
        END WHILE
        IF i == start                                    # empty label: "", "@", "a.", ".a", "a..b"
            RETURN false
        END IF
        labels = labels + 1
        IF i == n
            RETURN labels >= 2
        END IF
        IF namespace[i] != "."
            RETURN false
        END IF
        i = i + 1
    END LOOP
END FUNCTION

# STXT-SPEC §4.2 / §4.3: names contain Unicode letters, decimal digits and combining
# marks (Mn, Mc: Devanagari vowel signs, accents with no precomposed form...) plus the
# three ASCII separators, and at least one letter or digit. The character check is
# deliberately made AFTER NFC: the decomposed spelling "e" + combining acute is a valid
# spelling of "é" and must not be rejected before canonicalization.
CONSTANT NODE_NAME_FORMAT: REGEX = "^[\\p{L}\\p{Nd}\\p{Mn}\\p{Mc}\\-_ ]+$"
CONSTANT NODE_NAME_LETTER_OR_DIGIT: REGEX = "[\\p{L}\\p{Nd}]"

# Checks whether a logical node name is valid in a source document, schema or
# template. The caller chooses whether the resulting error is syntactic
# (ParseException) or semantic (ValidationException).
FUNCTION isValidNodeName(name: STRING): BOOLEAN

    IF name IS NULL
        RETURN false
    END IF

    logical_name: STRING = compactSpaces(name)
    nfc_name: STRING = normalizeNFC(logical_name)

    IF NOT NODE_NAME_FORMAT.matches(nfc_name)
        RETURN false
    END IF

    # At least one letter or digit (§4.2): a string made only of separators ("___") or
    # only of combining marks has no logical name.
    RETURN NODE_NAME_LETTER_OR_DIGIT.find(nfc_name)

END FUNCTION

# Validates the format of an already lower-cased namespace.
# An empty or NULL namespace is ignored (nodes without namespace are legal).
PROCEDURE validateNamespaceFormat(namespace: STRING, line_number: INTEGER)

    IF namespace IS NULL OR namespace.isEmpty()
        RETURN
    END IF

    IF NOT isValidNamespaceFormat(namespace)
        THROW ParseException(line_number, "INVALID_NAMESPACE", "Namespace not valid: " + namespace)
    END IF

END PROCEDURE
