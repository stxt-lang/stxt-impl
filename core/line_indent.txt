# LineIndent: a source line already split into its indentation and its content, as
# produced by parseLine(). It is what tells the Parser whether the line opens a node,
# continues a text block or is just a comment.
#
# parseLine() implements the indentation rules of STXT-SPEC §8 and the text block
# line rules of §10.2 / §10.3:
#   * 1 level = 1 TAB or TAB_SPACES spaces.
#   * The indentation of a single line MUST be homogeneous: only tabs or only spaces.
#     Mixing both in the same line is a parse error: INDENTATION_MIXED (§8.1, §8.3).
#   * Comment lines are validated like node lines (§9, §11): homogeneous indentation, a
#     multiple of TAB_SPACES when spaces, and a level of at most last_level + 1. They
#     produce no node and never move the hierarchy (the Parser does not update last_level
#     for them). Empty lines are never an error: their indentation is not validated.

CLASS LineIndent
    FIELD indent_level: INTEGER         # indentation level of the line
    FIELD line_without_indent: STRING   # content with the indentation already removed
    FIELD is_comment: BOOLEAN           # true if the line is a comment ('#')
    FIELD is_block: BOOLEAN             # true if the line is a text line of an open BLOCK node ('>>')
    FIELD content_start: INTEGER        # index of the first character of the content, i.e. the number of characters the indentation took up

    CONSTRUCTOR LineIndent(indent_level, line_without_indent, is_comment, is_block, content_start)
        indent_level        = indent_level
        line_without_indent = line_without_indent
        is_comment          = is_comment
        is_block            = is_block
        content_start       = content_start
    END CONSTRUCTOR

    # true if the line has no content beyond blanks (SPACE/TAB only, STXT-SPEC §4):
    # a line holding just an NBSP is NOT empty.
    FUNCTION isEmpty(): BOOLEAN
        RETURN trim(line_without_indent).isEmpty()
    END FUNCTION

END CLASS

FUNCTION parseLine(line: STRING, last_node_block: BOOLEAN, last_level: INTEGER, num_line: INTEGER): LineIndent

    # Indentation parsing state
    level: INTEGER = 0
    spaces: INTEGER = 0
    pointer: INTEGER = 0
    saw_space: BOOLEAN = false
    saw_tab: BOOLEAN = false
    is_comment: BOOLEAN = false

    WHILE pointer < line.length()

        c: CHAR = line.charAt(pointer)

        IF c == SPACE
            saw_space = true
            spaces = spaces + 1

            IF spaces == TAB_SPACES
                level = level + 1
                spaces = 0
            END IF

        ELSE IF c == TAB
            saw_tab = true
            level = level + 1
            spaces = 0

        ELSE IF c == COMMENT_CHAR
            # Comment line: produces no node, but its indentation is validated below exactly
            # like a node's (§9). Reached only when the line is not block text (a '#' deeper
            # than an open block is caught as text by the check below, before getting here),
            # so inside an open block a comment always has indent <= the block node: the
            # Parser closes the block (§9.1) and then hands the comment over to the observers
            # (Observer.onComment).
            is_comment = true
            BREAK

        ELSE
            # First non space/tab/comment character: end of indentation
            BREAK
        END IF

        # Inside a text block: the line belongs to the block as soon as its level
        # goes beyond the level of the block node. The rest of the line is free
        # text: only right trim is applied (§10.2).
        IF last_node_block AND level > last_level
            text: STRING = rightTrim(line.substring(pointer + 1))

            # The prefix covering the block level MUST be homogeneous (§10.2 rule 1).
            # Empty lines are always preserved and exempt from the check (§10.3).
            IF saw_space AND saw_tab AND text.length() > 0
                THROW ParseException(num_line, "INDENTATION_MIXED", "Mixed tabs and spaces in indentation")
            END IF

            # pointer is the index of the indentation character that crossed the block
            # level; the indentation took pointer + 1 characters
            RETURN LineIndent(level, text, false, true, pointer + 1)
        END IF

        pointer = pointer + 1

    END WHILE

    # At this point we are outside a text block (if any)

    # Empty or whitespace-only line
    IF pointer == line.length()
        IF last_node_block
            # Empty lines inside a block are preserved as "" (§10.3)
            RETURN LineIndent(level, "", false, true, pointer)
        ELSE
            # Empty lines outside a block are ignored; indentation not validated (§11)
            RETURN LineIndent(level, "", false, false, pointer)
        END IF
    END IF

    # Mixed tabs and spaces in the indentation of the same line (§8.1, §8.3)
    IF saw_space AND saw_tab
        THROW ParseException(num_line, "INDENTATION_MIXED", "Mixed tabs and spaces in indentation")
    END IF

    # Invalid indentation: spaces not a multiple of TAB_SPACES
    IF spaces > 0
        THROW ParseException(num_line, "INDENTATION_SPACES_NOT_VALID", "There are " + spaces + " spaces before node")
    END IF

    # Validate indentation level progression (no jumps, §11.3). Comments included (§9):
    # last_level is the level of the last NODE, a comment never becomes the reference.
    IF level > (last_level + 1)
        THROW ParseException(num_line, "INDENTATION_LEVEL_NOT_VALID", "Level of indent incorrect: " + level)
    END IF

    # Comment: the text after '#', verbatim
    IF is_comment
        RETURN LineIndent(level, line.substring(pointer + 1), true, false, pointer)
    END IF

    # General case: return the line without the consumed indentation
    # trim() is the blank-only trim of string_utils, never the platform one (§4)
    RETURN LineIndent(level, trim(line.substring(pointer)), false, false, pointer)

END FUNCTION
