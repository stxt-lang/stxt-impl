# Node of the STXT tree. It represents both INLINE nodes (with getValue()) and BLOCK
# text nodes (with getTextLines()), as told apart by isTextNode().
#
# Mutable while parsing (addChild() / addTextLine() are public); once the document is
# closed it MUST be treated as read-only.

CLASS Node

    # STXT-SPEC §4.2: Unicode letters and digits (categories L and Nd) plus '-', '_' and space
    CONSTANT VALID_NAME: REGEX = "^[\\p{L}\\p{Nd}\\-_ ]+$"

    # Basic fields
    FIELD line: INTEGER
    FIELD level: INTEGER
    FIELD name: STRING
    FIELD namespace: STRING
    FIELD block_text: BOOLEAN

    # Derived fields
    FIELD normalized_name: STRING
    FIELD qualified_name: STRING

    # Children and values
    FIELD value: STRING
    FIELD text_lines: STRING[] = []
    FIELD children: Node[] = []

    CONSTRUCTOR (line, level, name, namespace, block_text, value)
        line 		= line
        level 		= level
        name 		= compactSpaces(name)
        namespace 	= lowerCase(namespace)
        block_text	= block_text
        value 		= trimToNotNull(value)

        normalized_name	= normalizeChars(name)

        IF isEmpty(namespace)
            qualified_name = normalized_name
        ELSE
            qualified_name = namespace + ":" + normalized_name
        END IF

        # Basic checks

        # Namespace format ('a.b' minimum, ASCII, optional leading '@'). Throws if not valid.
        validateNamespaceFormat(namespace, line)

        # Safety net: a BLOCK node cannot carry an inline value (also checked by createNode)
        IF NOT isEmpty(value) AND block_text
            THROW RuntimeException("INLINE_VALUE_NOT_VALID", "Not empty value with block text")
        END IF

        # Allowed name characters (STXT-SPEC §4.2, error rule §11.8)
        IF NOT VALID_NAME.matches(name)
            THROW ParseException(line, "INVALID_NODE_NAME", "Node name contains invalid characters: " + name)
        END IF

        # An empty canonical name (e.g. "___") is invalid (STXT-SPEC §4.3, §11.7)
        IF isEmpty(normalized_name)
            THROW ParseException(line, "INVALID_NODE_NAME", "Node name not valid: " + name)
        END IF
    END CONSTRUCTOR

    # ----------------------------------------
    # Accessors
    # ----------------------------------------

    # Original name as it appears in the document (with spaces compacted)
    FUNCTION getName(): STRING
        RETURN name
    END FUNCTION

    # Canonical name (STXT-SPEC §4.3), used to compare and look up by structural identity
    FUNCTION getNormalizedName(): STRING
        RETURN normalized_name
    END FUNCTION

    # Effective namespace (own or inherited from the parent), lower-cased; "" if none
    FUNCTION getNamespace(): STRING
        RETURN namespace
    END FUNCTION

    # Logical identity of the node: namespace:canonical_name (or just the name if there is no namespace)
    FUNCTION getQualifiedName(): STRING
        RETURN qualified_name
    END FUNCTION

    # Inline value of an INLINE node, or "" for a BLOCK node
    FUNCTION getValue(): STRING
        RETURN value
    END FUNCTION

    # Text lines of a BLOCK node, in order of appearance
    FUNCTION getTextLines(): STRING[]
        RETURN text_lines
    END FUNCTION

    # Appends a text line to a BLOCK node
    PROCEDURE addTextLine(text_line: STRING)
        text_lines.push(text_line)
    END PROCEDURE

    # Children of the node in order of appearance
    FUNCTION getChildren(): Node[]
        RETURN children
    END FUNCTION

    # Appends an already closed child at the end of the list of children
    PROCEDURE addChild(node: Node)
        children.push(node)
    END PROCEDURE

    FUNCTION getLine(): INTEGER
        RETURN line
    END FUNCTION

    FUNCTION getLevel(): INTEGER
        RETURN level
    END FUNCTION

    # true if the node is a text block (BLOCK, '>>'); false if it is INLINE
    FUNCTION isTextNode(): BOOLEAN
        RETURN block_text
    END FUNCTION

    # Effective text: the block lines joined with '\n', or the inline value if not a block
    FUNCTION getText(): STRING
        IF isTextNode()
            RETURN join(text_lines, "\n")
        ELSE
            RETURN value
        END IF
    END FUNCTION

    # ----------------------------------------
    # Children lookup (by canonical name + namespace)
    # ----------------------------------------

    # Looks up the ONLY direct child with that canonical name in this node's own namespace.
    # Returns NULL if there is none; throws AMBIGUOUS_CHILD if there is more than one.
    FUNCTION getChild(cname: STRING): Node
        RETURN getChild(cname, namespace)
    END FUNCTION

    FUNCTION getChild(cname: STRING, child_namespace: STRING): Node
        result: Node[] = getChildren(cname, child_namespace)
        IF result.size() > 1
            THROW RuntimeException("AMBIGUOUS_CHILD", "More than 1 child. Use getChildren")
        END IF
        IF result.size() == 0
            RETURN NULL
        END IF
        RETURN result[0]
    END FUNCTION

    # Every direct child with that canonical name in this node's own namespace,
    # preserving the order of appearance.
    FUNCTION getChildren(cname: STRING): Node[]
        RETURN getChildren(cname, namespace)
    END FUNCTION

    FUNCTION getChildren(cname: STRING, child_namespace: STRING): Node[]
        key: STRING = normalizeChars(cname)
        result: Node[] = []
        FOREACH child IN children
            IF child.getNormalizedName() == key AND child.getNamespace() == child_namespace
                result.push(child)
            END IF
        END FOREACH
        RETURN result
    END FUNCTION

END CLASS
