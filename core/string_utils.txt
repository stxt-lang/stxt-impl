# String utilities. Unless stated otherwise, they take a STRING and return a STRING.

# Unicode NFC normalization. It unifies the precomposed and decomposed forms of the
# same character (for example, "é" and "e" + combining acute accent). A port MUST
# use Unicode NFC, independently of its current locale.
FUNCTION normalizeNFC (str)
END FUNCTION

# Trim + collapse of every inner whitespace run into a single space. Used for node names.
FUNCTION compactSpaces (str)
END FUNCTION

# Canonical node name (STXT-SPEC §4.3). Diacritics and non-Latin letters are KEPT:
# name equality is insensitive to case and separators, but sensitive to accents and
# alphabet, following the IDN model. E.g. "Año" == "año", but "Año" != "Ano".
#
# Process:
#   1. Trim
#   2. Unicode NFC normalization (precomposed and decomposed forms become equivalent)
#   3. Lower case (Unicode case folding)
#   4. Every run of separators ('-', '_', spaces) becomes a single '-'
#   5. Leading and trailing '-' are removed
#
# May return the empty string (e.g. for "___"): the caller MUST treat that as a
# parse error (STXT-SPEC §4.3, §11.7).
FUNCTION normalizeChars (str)
END FUNCTION

# Lower case (root locale / locale independent). Returns "" if str is NULL. Used for namespaces.
FUNCTION lowerCase (str)
END FUNCTION

# Returns "" if str is NULL; otherwise str trimmed on both sides.
FUNCTION trimToNotNull(str)
END FUNCTION

# Removes spaces and tabs at the end of the line only. Used for '>>' text block lines.
FUNCTION rightTrim (str)
END FUNCTION

# true if str is NULL or the empty string.
FUNCTION isEmpty(str)
END FUNCTION

# Removes the leading UTF-8 BOM if present.
FUNCTION removeUTF8BOM(str)
END FUNCTION

# Joins the strings of an array inserting the separator between them.
FUNCTION join(parts: STRING[], separator: STRING)
END FUNCTION

# Splits str by the separator and returns the fragments.
FUNCTION split(str: STRING, separator: STRING): STRING[]
END FUNCTION

# true if str ends with the given suffix.
FUNCTION endsWith(str: STRING, suffix: STRING): BOOLEAN
END FUNCTION

# Case-insensitive string comparison.
FUNCTION equalsIgnoreCase(a: STRING, b: STRING): BOOLEAN
END FUNCTION
