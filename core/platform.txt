# Platform: utilities that depend on the implementation language/platform.
# Every port MUST provide an equivalent of these functions.

# Splits the content into lines, accepting both "\n" and "\r\n" as line breaks (a lone
# "\r" is content, STXT-SPEC §3). The part after the last break is a line too, possibly
# empty: "a\r\nb\n" gives ["a", "b", ""]. Used by TextNode.setText().
FUNCTION splitLines(str: STRING): STRING[]
END FUNCTION

# Iterates the lines of the content LAZILY, with the same line breaks as splitLines, but
# without the trailing empty line: the final line break terminates the last line, so
# "a\nb\n" and "a\nb" both give "a", "b" (and "" gives no line). The parser consumes it
# (core/parser.txt parseResult): a port MUST NOT materialise every line before the parser
# has applied its limits to each. A port whose natural line source splits on more than
# LF/CRLF (BufferedReader.readLine, str.splitlines, String.lines) MUST NOT use it here.
FUNCTION lineIterator(str: STRING): ITERATOR<STRING>
END FUNCTION

# true if str represents a valid integer (sign allowed). Equivalent to "parseInteger would succeed".
FUNCTION isInteger(str: STRING): BOOLEAN
END FUNCTION

# true if str is a non-negative integer with no sign and no trailing text ("^\d+$").
# Used by the template cardinality parser.
FUNCTION isNatural(str: STRING): BOOLEAN
END FUNCTION

# Converts str into an INTEGER. MUST only be called when isInteger/isNatural is true.
FUNCTION parseInteger(str: STRING): INTEGER
END FUNCTION

# true if str is valid Base64 per STXT-SCHEMA-SPEC §9.5, that is, all of:
#   1. non-empty;
#   2. only the standard alphabet of RFC 4648 §4 ([A-Za-z0-9+/]; NOT the URL-safe '-' '_');
#   3. optional trailing padding: if present, exactly the one or two '=' that complete the
#      last quartet (so 'aGVsbG8=' and 'aGVsbG8' are valid, 'aGVsbG8==' and 'aGVsbG8x=' are
#      not), and only at the end;
#   4. no leftover bits: without the padding, length % 4 != 1, and the last character
#      encodes zero bits beyond the byte boundary (one of [AQgw] when length % 4 == 2,
#      one of [AEIMQUYcgkosw048] when length % 4 == 3); equivalently, decoding and
#      re-encoding (padding aside) gives the same string.
# The platform decoder is usually MORE lenient than this (ignoring invalid characters,
# accepting the empty string, or other paddings): the port MUST check these four points
# itself. The caller has already removed every blank (see binaryValue in schema/types.txt).
FUNCTION isValidBase64(str: STRING): BOOLEAN
END FUNCTION
