# Platform: utilities that depend on the implementation language/platform.
# Every port MUST provide an equivalent of these functions.

# Splits the content into lines, accepting both "\n" and "\r\n" as line breaks.
FUNCTION splitLines(str: STRING): STRING[]
END FUNCTION

# true if str represents a valid integer (sign allowed). Equivalent to "parseInteger would succeed".
FUNCTION isInteger(str: STRING): BOOLEAN
END FUNCTION

# Converts str into an INTEGER. MUST only be called when isInteger(str) is true.
FUNCTION parseInteger(str: STRING): INTEGER
END FUNCTION

# true if "min,max" are two valid integers. Used by the template cardinality parser.
FUNCTION isValidMinMax(str: STRING): BOOLEAN
END FUNCTION

# Parses a URI. Returns an object with getScheme()/getHost(), or NULL if not valid.
FUNCTION parseURI(str: STRING)
END FUNCTION

# true if str is decodable Base64 content.
FUNCTION isValidBase64(str: STRING): BOOLEAN
END FUNCTION
