# Language constants. Spec: STXT-SPEC §3 (encoding), §5 (':' inline nodes),
# §6 ('>>' block nodes), §8 (indentation) and §9 (comments).

COMMENT_CHAR 		= "#"
TAB_SPACES 			= 4
TAB 				= "\t"
SPACE 				= " "
SEP_NODE			= ":"
SEP_TEXT_NODE		= ">>"
DEFAULT_ENCODING	= "UTF-8"
EMPTY_NAMESPACE		= ""

# Cardinality bound (STXT-SCHEMA-SPEC §10, STXT-TEMPLATE-SPEC §7.1): Min/Max in a schema
# and the numbers of a template cardinality MUST NOT exceed 2^32 - 1; a greater value is
# CARDINALITY_NOT_VALID. Platform-independent: ports parse the number into a type that
# holds at least this value (long in Java, Number in JS, int in Python).
MAX_CARDINALITY     = 4294967295
# Digits of MAX_CARDINALITY: a longer numeral exceeds the bound before it is converted
# (schema/schema_parser.txt getInteger, template/child_line_parser.txt parseCount).
MAX_CARDINALITY_DIGITS = 10

# Parser limits (STXT-SPEC §11.2). The spec RECOMMENDS applying limits and these are its
# recommended defaults; the official ports enable them. A parser that applies limits MUST
# let the program configure them, and -1 disables the corresponding limit. Lengths are
# measured in the platform's natural string units (code points or UTF-16 units).
DEFAULT_MAX_NESTING     = 100        # max open levels (level 0 is the first)
DEFAULT_MAX_LINE_LENGTH = 10000      # max length of one input line, indentation included
DEFAULT_MAX_INPUT_SIZE  = 10000000   # max total input consumed

# Version of the STXT specifications (STXT-SPEC, STXT-TREE-SPEC, STXT-SCHEMA-SPEC,
# STXT-TEMPLATE-SPEC, STXT-DISCOVERY-SPEC) this implementation conforms to. It is NOT the
# version of the port's package: two ports with different package versions read the same
# STXT as long as this number is the same. Every port exposes it (since 0.10.0) and the
# CLI prints it with --version. Shared by the five specifications; major.minor, no patch.
SPEC_VERSION		= "1.0"
