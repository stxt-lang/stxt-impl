# Parser: line-by-line STXT parsing engine. It knows nothing about schemas: semantic
# validation is plugged in through registerValidator(), process observation through
# registerObserver() and result observation through registerStreamObserver()
# (see processors/).
#
# Algorithm: a stack of open nodes, one per indentation level (the size of the stack is
# the level of the next line). When a line at level N arrives, every node deeper than N
# is closed (validated and notified to the observers), and the new node is created,
# ATTACHED to its parent (the top of the stack; or kept as a root if the stack is empty)
# and pushed onto the stack, still open. Attaching at creation links both ends of the
# parent link (core/node.txt), so the node already knows its parent — and so its effective
# namespace and its level — when the observers see it in onCreate(). A document may hold
# multiple root nodes, which close under concatenation (STXT-SPEC §8.5 — logs/streaming
# friendly).
#
# Three entry points over the same traversal:
#   * parse():        traditional fail-fast, throws the first error found.
#   * parseResult():  multi-error, collects every error (syntax and validation) in a
#                     ParseResult without bailing out on the first one.
#   * parseStream():  input from a line iterator, and NOTHING retained: no nodes, no
#                     errors. Results reach the program only through the registered
#                     StreamObservers (each completed root by onRootNode(), each error
#                     by onError()), so memory holds one root tree at a time. This is
#                     the entry point for files that do not fit in memory.
#
# Which callbacks fire never depends on the entry point, only on what is registered:
# Observers, Validators and StreamObservers all fire in the three modes. In fail-fast
# parse() a StreamObserver still sees every error through onError() before the first
# one is thrown, because parse() reuses the parseResult() traversal.
#
# Parser limits (STXT-SPEC §11.2): the parser aborts on inputs that exceed the
# nesting depth, line length or input size limits, set to the DEFAULT_MAX_* values of
# core/constants.txt unless configured; -1 disables one. A limit error is a
# LimitException (exceptions/exceptions.txt) and ABORTS the parse in every mode: it is
# emitted (collected and/or notified) and nothing else is processed or notified — the
# nodes still open are not closed. Every port exposes the three limits with its own
# idiom (an options object, setters, keyword arguments), keeping these names
# recognizable: max_nesting, max_line_length, max_input_size. Each limit MUST be an
# integer >= 0 or -1; any other value is rejected when it is set (the platform's argument
# error), so that NaN cannot disable a limit silently nor -2 reject every line.

CLASS Parser

    FIELD observers: Observer[] = []
    FIELD stream_observers: StreamObserver[] = []
    FIELD validators: Validator[] = []

    # Parser limits (STXT-SPEC §11.2); -1 disables one.
    FIELD max_nesting: INTEGER     = DEFAULT_MAX_NESTING
    FIELD max_line_length: INTEGER = DEFAULT_MAX_LINE_LENGTH
    FIELD max_input_size: INTEGER  = DEFAULT_MAX_INPUT_SIZE

    # Registers an observer, notified when each node is opened and closed, and for
    # every comment and text line.
    PROCEDURE registerObserver(observer: Observer)
        observers.push(observer)
    END PROCEDURE

    # Registers a stream observer, notified with each completed root node and each
    # error, in every mode (see processors/stream_observer.txt).
    PROCEDURE registerStreamObserver(stream_observer: StreamObserver)
        stream_observers.push(stream_observer)
    END PROCEDURE

    # Registers a validator, invoked when each node is closed.
    PROCEDURE registerValidator(validator: Validator)
        validators.push(validator)
    END PROCEDURE

    # Traditional fail-fast mode: throws the first error found (either syntax or
    # validation). It reuses the same traversal as parseResult().
    FUNCTION parse(content: STRING): Node[]
        result: ParseResult = parseResult(content)

        IF result.hasErrors()
            THROW result.getErrors()[0]
        END IF

        RETURN result.getNodes()
    END FUNCTION

    # Multi-error mode: parses the whole content collecting every error found
    # (both syntax and validation) without bailing out on the first one — except a
    # LimitException, which aborts and is in every case the last error collected.
    FUNCTION parseResult(content: STRING): ParseResult
        result: ParseResult = ParseResult()

        # The lines are produced lazily (lineIterator in core/platform.txt), not split up
        # front: splitting the whole content materialises every line before the input-size
        # limit can act (§11.2: it is checked as the input is consumed), so an input far
        # above the limit cost memory proportional to its size instead of to the limit
        # (210 MB of input took 2.6 GB before aborting). The iterator also applies the
        # final-line-break rule: "a\n" is one line, not "a" plus an empty line (that would
        # add a spurious line to a '>>' block at EOF, STXT-SPEC §10.3).
        parseLines(lineIterator(content), result)

        RETURN result
    END FUNCTION

    # Streaming mode: input from a line iterator (each item one line, without its
    # line break; ports take their platform's natural line source — a lazy reader,
    # a generator, an iterable), and nothing retained. Results reach the program
    # only through the registered StreamObservers.
    PROCEDURE parseStream(lines: ITERATOR<STRING>)
        parseLines(lines, NULL)
    END PROCEDURE

    # Shared traversal. With a result, roots and errors are collected into it
    # (parse()/parseResult()); with NULL, nothing is retained (parseStream()).
    # Either way every registered callback fires the same.
    PROCEDURE parseLines(lines: ITERATOR<STRING>, result: ParseResult | NULL)
        stack: Node[] = []
        line_number: INTEGER = 0
        consumed: INTEGER = 0

        FOREACH line IN lines
            line_number = line_number + 1

            # A UTF-8 BOM only means anything at the very start of the input (STXT-SPEC §3)
            IF line_number == 1
                line = removeUTF8BOM(line)
            END IF

            # Limits first (STXT-SPEC §11.2): a limit error aborts, leaving the
            # open nodes unclosed and unnotified.
            IF max_line_length != -1 AND line.length() > max_line_length
                emitError(LimitException(line_number, "LIMIT_LINE_LENGTH_EXCEEDED",
                    "Line longer than " + max_line_length + " characters"), result)
                RETURN
            END IF

            consumed = consumed + line.length() + 1    # the line separator counts as one
            IF max_input_size != -1 AND consumed > max_input_size
                emitError(LimitException(line_number, "LIMIT_INPUT_SIZE_EXCEEDED",
                    "Input larger than " + max_input_size + " characters"), result)
                RETURN
            END IF

            IF NOT processLine(line, line_number, stack, result)
                RETURN    # a limit aborted the parse; its error is already emitted
            END IF
        END FOREACH

        # Close every node still open at EOF
        closeToLevel(stack, 0, result)
    END PROCEDURE

    # Processes one source line. Errors of this line are collected into the result and
    # the traversal continues with the next line: returns true to keep going, false when
    # a limit aborted the parse (its error is already emitted) — parseLines() stops on it.
    FUNCTION processLine(line_string, line_number, stack, result): BOOLEAN
        TRY
            last_node: Node = NULL
            IF NOT stack.isEmpty()
                last_node = stack.last()
            END IF

            # One open node per level: the top of the stack is at level size - 1. With no
            # open node the reference level is -1 (STXT-SPEC §8.3): the first line of the
            # document, and the first after every node has been closed, must be at level 0.
            last_level: INTEGER = -1
            IF last_node IS NOT NULL
                last_level = stack.size() - 1
            END IF

            last_node_block = last_node IS NOT NULL AND last_node IS TextNode

            # Split the line into indentation and content
            line_indent: LineIndent = parseLine(line_string, last_node_block, last_level, line_number)

            # Comment line: produces no node. Its indentation was validated by parseLine like
            # a node's (§9), but it never becomes the reference level: last_level is only
            # updated by nodes. A comment with indentation <= the level of an open BLOCK node
            # closes that block (STXT-SPEC §6.1, §9.1): a block is a literal and cannot be
            # commented from inside. Only the block closes; the comment does not touch the
            # rest of the hierarchy.
            IF line_indent.is_comment
                IF last_node_block
                    closeToLevel(stack, stack.size() - 1, result)
                END IF

                FOREACH observer IN observers
                    observer.onComment(line_number, line_string)
                END FOREACH
                RETURN true
            END IF

            # Text line of an open BLOCK node: append it instead of creating a node
            IF line_indent.is_block
                text_node: TextNode = last_node
                text_node.addTextLine(line_indent.line_without_indent)

                FOREACH observer IN observers
                    observer.onTextLine(text_node, line_number, line_string, line_indent)
                END FOREACH
                RETURN true
            END IF

            # Empty lines outside a block are ignored
            IF line_indent.isEmpty()
                RETURN true
            END IF

            current_level = line_indent.indent_level

            # Nesting limit (STXT-SPEC §11.2): only a node line can open a new level.
            # Comment and block text lines returned above; with the consecutive-level
            # rule this triggers exactly when the first node at level max_nesting opens.
            IF max_nesting != -1 AND current_level >= max_nesting
                emitError(LimitException(line_number, "LIMIT_NESTING_EXCEEDED",
                    "Nesting deeper than " + max_nesting + " levels"), result)
                RETURN false
            END IF

            # Close nodes down to the current level (finalizes them)
            closeToLevel(stack, current_level, result)

            # Create the new node, attach it to its parent (or keep it as a root if the
            # stack is empty) and keep it open on the stack. The parent is always an
            # InlineNode: a TextNode on top of the stack only takes text lines (is_block
            # above), and any line at its level or shallower has just closed it.
            node: Node = createNode(line_indent, line_number)

            IF NOT stack.isEmpty()
                parent: InlineNode = stack.last()
                parent.addChild(node)
            END IF

            # Hand it over to the observers (the node already has its parent)
            FOREACH observer IN observers
                observer.onCreate(node, line_string)
            END FOREACH

            stack.push(node)

        CATCH pe: ParseException
            emitError(pe, result)
        CATCH e
            # Unexpected platform error: collected as a ParseException
            emitError(ParseException(line_number, "UNEXPECTED_ERROR", e.getMessage()), result)
        END TRY

        RETURN true
    END FUNCTION

    # Closes every node deeper than target_level: runs the validators over it and notifies
    # the observers. (The node was attached to its parent when it was created.) When the
    # node closed is a root — the stack is left empty — the StreamObservers receive it
    # complete by onRootNode(), and it is collected into the result if there is one
    # (parse()/parseResult()) or released (parseStream()).
    PROCEDURE closeToLevel(stack: Node[], target_level: INTEGER, result: ParseResult | NULL)

        WHILE stack.size() > target_level

            completed: Node = stack.pop()

            # A closing BLOCK node drops its final empty lines (STXT-SPEC §10.3): they are
            # not content, only visual separation or an editor's final line breaks. The
            # validators and observers below already see the trimmed node. Observer
            # callbacks are process observation and fired as the lines were read, so
            # onTextLine() did fire for these lines while the block was open.
            IF completed IS TextNode
                completed.removeTrailingEmptyLines()
            END IF

            # Hand it over to the validators: they return errors, they do not throw
            FOREACH validator IN validators
                TRY
                    errors: ValidationException[] = validator.validate(completed)
                    FOREACH error IN errors
                        emitError(error, result)
                    END FOREACH
                CATCH pe: ParseException
                    emitError(pe, result)
                CATCH e
                    # A validator that throws unexpectedly does not abort the parse
                    emitError(ParseException(completed.getLine(), "UNEXPECTED_ERROR", e.getMessage()), result)
                END TRY
            END FOREACH

            # Hand it over to the observers
            FOREACH observer IN observers
                observer.onFinish(completed)
            END FOREACH

            # A closed root: the stream observers receive it, the result collects it
            IF stack.isEmpty()
                FOREACH stream_observer IN stream_observers
                    stream_observer.onRootNode(completed)
                END FOREACH

                IF result IS NOT NULL
                    result.addNode(completed)
                END IF
            END IF

        END WHILE

    END PROCEDURE

    # Every error goes through here: collected into the result when there is one, and
    # notified to the StreamObservers always, in order of appearance.
    PROCEDURE emitError(error: ParseException, result: ParseResult | NULL)
        IF result IS NOT NULL
            result.addError(error)
        END IF

        FOREACH stream_observer IN stream_observers
            stream_observer.onError(error)
        END FOREACH
    END PROCEDURE

END CLASS

# Builds the node a line opens, telling apart the INLINE form ("Name: value") from the
# BLOCK one ("Name >>"). The node gets the namespace the LINE declares, if any; inheritance
# from the parent is resolved by the node itself through its parent link, once attached
# (Node.getNamespace()). No level either: it is derived from the parent chain.
FUNCTION createNode(line_indent: LineIndent, line_number: INTEGER): Node

    line: STRING = line_indent.line_without_indent

    name = NULL
    value = NULL
    is_text_node = false

    node_index: INTEGER = line.indexOf(SEP_NODE)
    text_index: INTEGER = line.indexOf(SEP_TEXT_NODE)

    IF node_index == -1 AND text_index == -1
        THROW ParseException(line_number, "INVALID_LINE", "Line not valid: " + line)

    ELSE IF node_index == -1 AND text_index != -1
        is_text_node = true

    ELSE IF node_index != -1 AND text_index == -1
        is_text_node = false

    ELSE IF node_index < text_index
        is_text_node = false

    ELSE IF node_index >= text_index
        THROW ParseException(line_number, "INVALID_LINE", "Line not valid: " + line)
    END IF

    IF is_text_node
        name = line.substring(0, text_index)
        value = line.substring(text_index + SEP_TEXT_NODE.length())
    ELSE
        name = line.substring(0, node_index)
        value = line.substring(node_index + SEP_NODE.length())
    END IF

    # A '>>' node cannot carry significant inline content on the same line (§11.4)
    IF is_text_node AND NOT trim(value).isEmpty()    # blank-only trim: '>>' + NBSP is an error
        THROW ParseException(line_number, "BLOCK_VALUE_NOT_ALLOWED", "Line not valid: " + line)
    END IF

    # The namespace the line declares, if any ("" when it inherits)
    nn: NameNamespace = NameNamespaceParser.parse(name, NULL, line_number, line)

    name = nn.getName()
    namespace: STRING = nn.getNamespace()

    # Validate the name
    IF name.isEmpty()
        THROW ParseException(line_number, "INVALID_LINE", "Line not valid: " + line)
    END IF

    IF is_text_node
        RETURN TextNode(name, namespace, NULL, line_number)
    ELSE
        RETURN InlineNode(name, namespace, value, line_number)
    END IF

END FUNCTION
