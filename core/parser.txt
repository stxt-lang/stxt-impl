# Parser: line-by-line STXT parsing engine. It knows nothing about schemas: semantic
# validation is plugged in through registerValidator() and observation through
# registerObserver() (see processors/).
#
# Algorithm: a stack of open nodes, one per indentation level (the size of the stack is
# the level of the next line). When a line at level N arrives, every node deeper than N
# is closed (validated and notified to the observers), and the new node is created,
# ATTACHED to its parent (the top of the stack; or added to the documents if it is a root)
# and pushed onto the stack, still open. Attaching at creation links both ends of the
# parent link (core/node.txt), so the node already knows its parent — and so its effective
# namespace and its level — when the observers see it in onCreate(). A document may hold
# multiple root nodes, which close under concatenation (STXT-SPEC §8.5 — logs/streaming
# friendly).
#
# Two modes:
#   * parse():       traditional fail-fast, throws the first error found.
#   * parseResult(): multi-error, collects every error (syntax and validation) in a
#                    ParseResult without bailing out on the first one.

CLASS Parser

    FIELD observers: Observer[] = []
    FIELD validators: Validator[] = []

    # Registers an observer, notified when each node is opened and closed, and for
    # every comment and text line.
    PROCEDURE registerObserver(observer: Observer)
        observers.push(observer)
    END PROCEDURE

    # Registers a validator, invoked when each node is closed.
    PROCEDURE registerValidator(validator: Validator)
        validators.push(validator)
    END PROCEDURE

    # Traditional fail-fast mode: throws the first error found (either syntax or
    # validation). It reuses the same traversal as parseResult().
    FUNCTION parse(content: STRING): Node[]
        result: ParseResult = parseResult(content)

        IF result.hasErrors()
            THROW result.getErrors()[0]
        END IF

        RETURN result.getNodes()
    END FUNCTION

    # Multi-error mode: parses the whole content collecting every error found
    # (both syntax and validation) without bailing out on the first one.
    FUNCTION parseResult(content: STRING): ParseResult
        content = removeUTF8BOM(content)

        result: ParseResult = ParseResult()
        stack: Node[] = []
        documents: Node[] = []
        line_number: INTEGER = 0

        lines: STRING[] = splitLines(content)

        # The final line break terminates the last line, it is not an extra empty line
        # (this avoids adding a spurious line to a '>>' block at EOF, STXT-SPEC §10.3)
        IF lines.length() > 0 AND lines.last() == ""
            lines.pop()
        END IF

        FOREACH line IN lines
            line_number = line_number + 1
            processLine(line, line_number, stack, documents, result)
        END FOREACH

        # Close every node still open at EOF
        closeToLevel(stack, 0, result)

        # Add the root nodes to the result
        FOREACH doc IN documents
            result.addNode(doc)
        END FOREACH

        RETURN result
    END FUNCTION

    # Processes one source line. Errors of this line are collected into the result;
    # the traversal continues with the next line.
    PROCEDURE processLine(line_string, line_number, stack, documents, result)
        TRY
            last_node: Node = NULL
            IF NOT stack.isEmpty()
                last_node = stack.last()
            END IF

            # One open node per level: the top of the stack is at level size - 1
            last_level: INTEGER = 0
            IF last_node IS NOT NULL
                last_level = stack.size() - 1
            END IF

            last_node_block = last_node IS NOT NULL AND last_node IS TextNode

            # Split the line into indentation and content
            line_indent: LineIndent = parseLine(line_string, last_node_block, last_level, line_number)

            # Comment line: produces no node. Its indentation was validated by parseLine like
            # a node's (§9), but it never becomes the reference level: last_level is only
            # updated by nodes. A comment with indentation <= the level of an open BLOCK node
            # closes that block (STXT-SPEC §6.1, §9.1): a block is a literal and cannot be
            # commented from inside. Only the block closes; the comment does not touch the
            # rest of the hierarchy.
            IF line_indent.is_comment
                IF last_node_block
                    closeToLevel(stack, stack.size() - 1, result)
                END IF

                FOREACH observer IN observers
                    observer.onComment(line_number, line_string)
                END FOREACH
                RETURN
            END IF

            # Text line of an open BLOCK node: append it instead of creating a node
            IF line_indent.is_block
                text_node: TextNode = last_node
                text_node.addTextLine(line_indent.line_without_indent)

                FOREACH observer IN observers
                    observer.onTextLine(text_node, line_number, line_string, line_indent)
                END FOREACH
                RETURN
            END IF

            # Empty lines outside a block are ignored
            IF line_indent.isEmpty()
                RETURN
            END IF

            current_level = line_indent.indent_level

            # Close nodes down to the current level (finalizes them)
            closeToLevel(stack, current_level, result)

            # Create the new node, attach it to its parent (or to the documents if it is a
            # root) and keep it open on the stack. The parent is always an InlineNode: a
            # TextNode on top of the stack only takes text lines (is_block above), and any
            # line at its level or shallower has just closed it.
            node: Node = createNode(line_indent, line_number)

            IF stack.isEmpty()
                documents.push(node)
            ELSE
                parent: InlineNode = stack.last()
                parent.addChild(node)
            END IF

            # Hand it over to the observers (the node already has its parent)
            FOREACH observer IN observers
                observer.onCreate(node, line_string)
            END FOREACH

            stack.push(node)

        CATCH pe: ParseException
            result.addError(pe)
        CATCH e
            # Unexpected platform error: collected as a ParseException
            result.addError(ParseException(line_number, "UNEXPECTED_ERROR", e.getMessage()))
        END TRY
    END PROCEDURE

    # Closes every node deeper than target_level: runs the validators over it and notifies
    # the observers. (The node was attached to its parent when it was created.)
    PROCEDURE closeToLevel(stack: Node[], target_level: INTEGER, result: ParseResult)

        WHILE stack.size() > target_level

            completed: Node = stack.pop()

            # Hand it over to the validators: they return errors, they do not throw
            FOREACH validator IN validators
                TRY
                    errors: ValidationException[] = validator.validate(completed)
                    FOREACH error IN errors
                        result.addError(error)
                    END FOREACH
                CATCH pe: ParseException
                    result.addError(pe)
                CATCH e
                    # A validator that throws unexpectedly does not abort the parse
                    result.addError(ParseException(completed.getLine(), "UNEXPECTED_ERROR", e.getMessage()))
                END TRY
            END FOREACH

            # Hand it over to the observers
            FOREACH observer IN observers
                observer.onFinish(completed)
            END FOREACH

        END WHILE

    END PROCEDURE

END CLASS

# Builds the node a line opens, telling apart the INLINE form ("Name: value") from the
# BLOCK one ("Name >>"). The node gets the namespace the LINE declares, if any; inheritance
# from the parent is resolved by the node itself through its parent link, once attached
# (Node.getNamespace()). No level either: it is derived from the parent chain.
FUNCTION createNode(line_indent: LineIndent, line_number: INTEGER): Node

    line: STRING = line_indent.line_without_indent

    name = NULL
    value = NULL
    is_text_node = false

    node_index: INTEGER = line.indexOf(SEP_NODE)
    text_index: INTEGER = line.indexOf(SEP_TEXT_NODE)

    IF node_index == -1 AND text_index == -1
        THROW ParseException(line_number, "INVALID_LINE", "Line not valid: " + line)

    ELSE IF node_index == -1 AND text_index != -1
        is_text_node = true

    ELSE IF node_index != -1 AND text_index == -1
        is_text_node = false

    ELSE IF node_index < text_index
        is_text_node = false

    ELSE IF node_index >= text_index
        THROW ParseException(line_number, "INVALID_LINE", "Line not valid: " + line)
    END IF

    IF is_text_node
        name = line.substring(0, text_index)
        value = line.substring(text_index + SEP_TEXT_NODE.length())
    ELSE
        name = line.substring(0, node_index)
        value = line.substring(node_index + SEP_NODE.length())
    END IF

    # A '>>' node cannot carry significant inline content on the same line (§11.4)
    IF is_text_node AND NOT trim(value).isEmpty()    # blank-only trim: '>>' + NBSP is an error
        THROW ParseException(line_number, "BLOCK_VALUE_NOT_ALLOWED", "Line not valid: " + line)
    END IF

    # The namespace the line declares, if any ("" when it inherits)
    nn: NameNamespace = NameNamespaceParser.parse(name, NULL, line_number, line)

    name = nn.getName()
    namespace: STRING = nn.getNamespace()

    # Validate the name
    IF name.isEmpty()
        THROW ParseException(line_number, "INVALID_LINE", "Line not valid: " + line)
    END IF

    IF is_text_node
        RETURN TextNode(name, namespace, NULL, line_number)
    ELSE
        RETURN InlineNode(name, namespace, value, line_number)
    END IF

END FUNCTION
