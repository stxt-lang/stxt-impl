# NameNamespace: the (name, namespace) pair resulting from interpreting the text that
# precedes ':' or '>>' in a node line. The namespace is always returned lower-cased.

CLASS NameNamespace
    FIELD name: STRING
    FIELD namespace: STRING

    CONSTRUCTOR (name, namespace)
        name = name
        namespace = namespace
    END CONSTRUCTOR

    FUNCTION getName(): STRING
        RETURN name
    END FUNCTION

    FUNCTION getNamespace(): STRING
        RETURN namespace
    END FUNCTION
END CLASS

# Splits the node name from its optional namespace in parentheses.
#
# Input (raw_name): the text before ':' or '>>', e.g.:
#   "Node name"
#   "Node name (com.example.docs)"
#   "Node name (@a.special.namespace)"
#
# The FORMAT of the namespace ('a.b' minimum, ASCII, optional '@') is NOT validated
# here; it is validated later when the Node is built (see validations.txt / node.txt).
# This function only splits the two parts and lower-cases the namespace.
FUNCTION parse(raw_name: STRING, inherited_namespace: STRING, line_number: INTEGER, full_line: STRING): NameNamespace

    IF raw_name IS NULL
        THROW ParseException(line_number, "INVALID_LINE", "Line not valid: " + full_line)
    END IF

    raw_name = trim(raw_name)    # blank-only trim (string_utils, STXT-SPEC §4)

    index_open: INTEGER  = raw_name.indexOf("(")
    index_close: INTEGER = raw_name.indexOf(")")

    name: STRING = NULL

    # Default namespace: the one inherited from the parent, or empty if there is none
    namespace: STRING = inherited_namespace
    IF namespace IS NULL
        namespace = EMPTY_NAMESPACE
    END IF

    IF index_open != -1 AND index_close != -1
        # Parentheses present: '(' must come before ')' and ')' must be the last character
        IF index_open > index_close OR index_close != raw_name.length() - 1
            THROW ParseException(line_number, "INVALID_NAMESPACE", "Line not valid: " + full_line)
        END IF

        name = trim(raw_name.substring(0, index_open))

        # NO trim here: the grammar (STXT-SPEC §7 / §16) does not allow spaces inside '( )',
        # so "( com.example )" must fail the namespace format validation later.
        namespace = raw_name.substring(index_open + 1, index_close)

        # The empty namespace "()" is not allowed
        IF namespace.isEmpty()
            THROW ParseException(line_number, "INVALID_NAMESPACE", "Line not valid: " + full_line)
        END IF

    ELSE IF index_open == -1 AND index_close == -1
        # No parentheses: the whole text is the name; the inherited namespace stands
        name = raw_name

    ELSE
        # A single parenthesis (open without close or close without open): invalid
        THROW ParseException(line_number, "INVALID_NAMESPACE", "Line not valid: " + full_line)
    END IF

    RETURN NameNamespace(name, lowerCase(namespace))

END FUNCTION
